\chapter{Réactions chimiques et loi de Lavoisier}

\noindent\textit{Quand du fer rouille, quand du bois brûle, quand du lait tourne : la
matière se transforme. Ce chapitre explique comment écrire et équilibrer ces
transformations, et pourquoi la masse totale ne change jamais.}
\vspace{8pt}

\connecteBanner

\section{Réaction chimique : réactifs et produits}

\begin{definition}[Réaction chimique]
Une \textbf{réaction chimique} est une transformation au cours de laquelle des
\textbf{réactifs} (les corps de départ) se transforment en \textbf{produits} (les corps
obtenus). On la représente par une \textbf{équation-bilan} :
\[
\text{réactifs}\longrightarrow\text{produits}.
\]
\end{definition}

\begin{illus}
\begin{tikzpicture}[every node/.style={font=\small}]
\node[draw=onbo,fill=onbsoft,rounded corners=4pt,minimum width=3cm,minimum height=1.2cm] (r) {Réactifs};
\node[draw=onbgreen,fill=onbgreen!10,rounded corners=4pt,minimum width=3cm,minimum height=1.2cm,right=3.4cm of r] (p) {Produits};
\draw[->,onbmuted,line width=1.2pt] (r) -- node[above]{réaction chimique} (p);
\end{tikzpicture}
\fig{Figure — Une réaction chimique transforme des réactifs en produits.}
\end{illus}

\begin{exemple}
Le fer et le soufre réagissent pour donner le sulfure de fer :
$\mathrm{Fe}+\mathrm{S}\longrightarrow\mathrm{FeS}$. Réactifs : fer et soufre. Produit :
sulfure de fer.
\end{exemple}

\section{Loi de Lavoisier : conservation de la masse}

\begin{propriete}[Loi de Lavoisier]
Au cours d'une réaction chimique, la masse totale des réactifs consommés est
\textbf{égale} à la masse totale des produits formés. Les atomes ne disparaissent pas et
ne se créent pas : ils se \textbf{réorganisent}. On résume souvent cette loi par : « Rien
ne se perd, rien ne se crée, tout se transforme. »
\end{propriete}

\begin{illus}
\begin{tikzpicture}[scale=1]
\draw[onbmuted,line width=2pt] (2,0) -- (2,0.3);
\draw[onbmuted,line width=1.5pt] (0,1.6) -- (4,1.6);
\draw[onbmuted,line width=1pt] (2,0.3) -- (2,1.6);
\draw[onbmuted,line width=1pt] (0,1.6) -- (0,0.9);
\draw[onbmuted,line width=1pt] (4,1.6) -- (4,0.9);
\draw[onbo,line width=1.2pt,fill=onbsoft] (-0.7,0.5) -- (0.7,0.5) -- (0.5,0.9) -- (-0.5,0.9) -- cycle;
\draw[onbgreen,line width=1.2pt,fill=onbgreen!10] (3.3,0.5) -- (4.7,0.5) -- (4.5,0.9) -- (3.5,0.9) -- cycle;
\node[below] at (0,0.35) {\small réactifs};
\node[below] at (4,0.35) {\small produits};
\node[above] at (2,1.9) {\small même masse totale};
\end{tikzpicture}
\fig{Figure — La masse totale des réactifs égale la masse totale des produits.}
\end{illus}

\begin{exemple}
$7$ g de fer réagissent avec $4$ g de soufre : $\mathrm{Fe}+\mathrm{S}\to\mathrm{FeS}$. La
masse de sulfure de fer obtenue est $7+4=11$ g.
\end{exemple}

\begin{remarque}
La loi de Lavoisier permet de calculer la masse d'un réactif ou d'un produit sans le
peser, à condition de connaître les masses des deux autres corps impliqués dans la
réaction.
\end{remarque}

\section{Équation-bilan et équilibrage}

\begin{propriete}[Équilibrer une équation-bilan]
Une équation-bilan est \textbf{équilibrée} lorsque l'on trouve le \textbf{même nombre}
d'atomes de chaque élément de part et d'autre de la flèche. Pour l'équilibrer, on ajoute
des \textbf{nombres} (appelés coefficients stœchiométriques) devant les formules, sans
jamais modifier les formules elles-mêmes.
\end{propriete}

\begin{illus}
\begin{tikzpicture}[every node/.style={font=\small},align=center]
\node[draw=onbo,rounded corners=3pt,fill=onbsoft,minimum width=3.6cm,minimum height=1.5cm] (L)
{$2\mathrm{H_2}+\mathrm{O_2}$\\[3pt]\footnotesize $4$ atomes H, $2$ atomes O};
\node[right=1.1cm of L,font=\Large] {$\longrightarrow$};
\node[draw=onbgreen,rounded corners=3pt,fill=onbgreen!10,minimum width=3.6cm,minimum height=1.5cm,right=4.2cm of L] (R)
{$2\mathrm{H_2O}$\\[3pt]\footnotesize $4$ atomes H, $2$ atomes O};
\end{tikzpicture}
\fig{Figure — L'équation $2\mathrm{H_2}+\mathrm{O_2}\to2\mathrm{H_2O}$ est équilibrée :
autant d'atomes H et O de chaque côté.}
\end{illus}

\begin{exemple}
Équilibrer $\mathrm{H_2}+\mathrm{Cl_2}\to\mathrm{HCl}$ : à droite, $1$ atome H et $1$
atome Cl seulement, contre $2$ et $2$ à gauche. On place un $2$ devant $\mathrm{HCl}$ :
$\mathrm{H_2}+\mathrm{Cl_2}\to2\mathrm{HCl}$. Il y a maintenant $2$ atomes H et $2$ atomes
Cl de chaque côté : l'équation est équilibrée.
\end{exemple}

\begin{remarque}
On ne modifie \textbf{jamais} les indices à l'intérieur d'une formule brute (par exemple
transformer $\mathrm{H_2O}$ en $\mathrm{H_2O_2}$) : cela changerait la nature du corps.
Seuls les coefficients \textbf{devant} les formules peuvent être modifiés.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Réaction chimique.} Réactifs $\to$ Produits, décrite par une équation-bilan.\\[4pt]
\textbf{Loi de Lavoisier.} Masse totale des réactifs $=$ masse totale des produits (rien
ne se perd, rien ne se crée).\\[4pt]
\textbf{Équation équilibrée.} Même nombre d'atomes de chaque élément de part et d'autre de
la flèche.\\[4pt]
\textbf{Coefficients.} On équilibre en ajoutant des nombres \textbf{devant} les formules,
jamais en modifiant les indices \textbf{dans} les formules.\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre le coefficient devant une formule (ex. le « $2$ »
de $2\mathrm{H_2O}$, qui multiplie toute la molécule) avec un indice à l'intérieur d'une
formule.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Identifier réactifs et produits dans un énoncé}
Repérer, dans le texte, les corps présents \textbf{avant} la réaction (réactifs) et les
corps obtenus \textbf{après} (produits), puis les placer de part et d'autre de la flèche.
\begin{exemple}
« Le carbone brûle dans le dioxygène pour donner du dioxyde de carbone » : réactifs
$\mathrm{C}$ et $\mathrm{O_2}$, produit $\mathrm{CO_2}$.
\end{exemple}

\methode{Appliquer la loi de Lavoisier pour calculer une masse manquante}
Écrire l'égalité (masse des réactifs) $=$ (masse des produits), puis isoler la masse
inconnue.
\begin{exemple}
$3$ g de carbone réagissent avec du dioxygène pour donner $11$ g de dioxyde de carbone :
masse de dioxygène consommée $=11-3=8$ g.
\end{exemple}

\methode{Vérifier qu'une équation-bilan est équilibrée}
Compter, pour chaque élément, le nombre total d'atomes à gauche puis à droite de la
flèche (en tenant compte des coefficients et des indices) : ils doivent être égaux.
\begin{exemple}
$\mathrm{Fe}+\mathrm{S}\to\mathrm{FeS}$ : $1$ atome Fe et $1$ atome S de chaque côté :
équation déjà équilibrée.
\end{exemple}

\methode{Équilibrer une équation-bilan simple}
Ajuster les coefficients devant les formules, un élément à la fois, jusqu'à obtenir le
même nombre d'atomes de chaque élément des deux côtés.
\begin{exemple}
$\mathrm{N_2}+\mathrm{H_2}\to\mathrm{NH_3}$ : équilibrer l'azote et l'hydrogène donne
$\mathrm{N_2}+3\mathrm{H_2}\to2\mathrm{NH_3}$ ($2$ atomes N et $6$ atomes H de chaque
côté).
\end{exemple}

\methode{Résoudre un problème concret avec la loi de Lavoisier}
Identifier la réaction en jeu, repérer les masses connues et la masse recherchée, puis
appliquer la conservation de la masse.
\begin{exemple}
$14$ g de fer réagissent entièrement avec $8$ g de soufre : la masse de sulfure de fer
obtenue est $14+8=22$ g.
\end{exemple}
