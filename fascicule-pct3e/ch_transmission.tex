\chapter{Transmission du mouvement de rotation}

\noindent\textit{Vélo, machine à laver, perceuse : de nombreux systèmes techniques
transmettent un mouvement de rotation d'un arbre à un autre, souvent en modifiant sa
vitesse. Ce chapitre présente les principaux systèmes de transmission et leur rapport de
transmission.}
\vspace{8pt}

\connecteBanner

\section{Les systèmes de transmission}

\begin{definition}[Roue menante, roue menée]
Dans un système de transmission, la \textbf{roue menante} (ou pignon menant) reçoit le
mouvement d'un moteur ; la \textbf{roue menée} reçoit ce mouvement par l'intermédiaire du
système de transmission.
\end{definition}

\begin{propriete}[Trois systèmes principaux]
\begin{itemize}
\item \textbf{Engrenage} : deux roues dentées en contact direct. Les sens de rotation des
deux roues sont \textbf{opposés}.
\item \textbf{Poulies-courroie} : deux poulies reliées par une courroie. Les sens de
rotation sont \textbf{identiques} ; un léger glissement de la courroie est possible.
\item \textbf{Chaîne-pignon} (ex. : vélo) : deux pignons dentés reliés par une chaîne. Les
sens de rotation sont \textbf{identiques}, sans glissement.
\end{itemize}
\end{propriete}

\section{Le rapport de transmission}

\begin{propriete}[Engrenage : vitesses et nombres de dents]
Pour un engrenage, le produit de la vitesse de rotation par le nombre de dents est le
même pour les deux roues :
\[
N_1\times Z_1=N_2\times Z_2 \qquad\text{soit}\qquad N_2=N_1\times\frac{Z_1}{Z_2},
\]
où $N_1$, $N_2$ sont les vitesses de rotation (en tr/min) et $Z_1$, $Z_2$ les nombres de
dents des roues menante et menée.
\end{propriete}

\begin{exemple}
Une roue menante tourne à $N_1=1\,000$ tr/min avec $Z_1=20$ dents ; la roue menée a
$Z_2=40$ dents : $N_2=1\,000\times\dfrac{20}{40}=500$ tr/min.
\end{exemple}

\begin{propriete}[Poulies-courroie : vitesses et diamètres]
Pour un système poulies-courroie (sans glissement), le produit de la vitesse de rotation
par le diamètre est le même pour les deux poulies :
\[
N_1\times D_1=N_2\times D_2 \qquad\text{soit}\qquad N_2=N_1\times\frac{D_1}{D_2}.
\]
\end{propriete}

\begin{exemple}
Une poulie menante de diamètre $D_1=50$ mm tourne à $N_1=1\,000$ tr/min ; la poulie
menée a $D_2=100$ mm : $N_2=1\,000\times\dfrac{50}{100}=500$ tr/min.
\end{exemple}

\begin{definition}[Rapport de transmission]
Le \textbf{rapport de transmission} $r$ vaut $r=\dfrac{N_2}{N_1}=\dfrac{Z_1}{Z_2}$ (ou
$\dfrac{D_1}{D_2}$). Si $r<1$, le système est \textbf{réducteur de vitesse} (la roue
menée tourne moins vite, mais avec plus de force) ; si $r>1$, il est
\textbf{multiplicateur de vitesse}.
\end{definition}

\begin{exemple}
Une roue menante de $Z_1=30$ dents entraîne une roue menée de $Z_2=20$ dents :
$r=\dfrac{30}{20}=1{,}5>1$ : le système est multiplicateur de vitesse.
\end{exemple}

\begin{remarque}
Un réducteur de vitesse (grande roue menée) permet d'obtenir un couple plus important :
c'est pourquoi les treuils, monte-charges et perceuses utilisent souvent des trains
d'engrenages réducteurs.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Engrenage.} Contact direct, sens \textbf{opposés}, $N_1Z_1=N_2Z_2$.\\[4pt]
\textbf{Poulies-courroie.} Sens \textbf{identiques}, glissement possible,
$N_1D_1=N_2D_2$.\\[4pt]
\textbf{Chaîne-pignon.} Sens identiques, sans glissement (comme l'engrenage pour les
vitesses, avec $N_1Z_1=N_2Z_2$).\\[4pt]
\textbf{Rapport de transmission.} $r=\dfrac{N_2}{N_1}$ : $r<1$ réducteur, $r>1$
multiplicateur.\\[4pt]
\textbf{Piège fréquent.} Dans un engrenage, la roue avec \textbf{le plus de dents} tourne
\textbf{le moins vite}.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer la vitesse de sortie d'un engrenage}
Appliquer $N_2=N_1\times\dfrac{Z_1}{Z_2}$.
\begin{exemple}
$N_1=1\,500$ tr/min, $Z_1=15$, $Z_2=45$ : $N_2=1\,500\times\dfrac{15}{45}=500$ tr/min.
\end{exemple}

\methode{Calculer la vitesse de sortie d'un système poulies-courroie}
Appliquer $N_2=N_1\times\dfrac{D_1}{D_2}$.
\begin{exemple}
$N_1=1\,200$ tr/min, $D_1=80$ mm, $D_2=40$ mm : $N_2=1\,200\times\dfrac{80}{40}=2\,400$
tr/min.
\end{exemple}

\methode{Déterminer si un système est réducteur ou multiplicateur}
Calculer $r=\dfrac{Z_1}{Z_2}$ (ou $\dfrac{D_1}{D_2}$) : $r<1$ réducteur, $r>1$
multiplicateur.
\begin{exemple}
$Z_1=10$, $Z_2=50$ : $r=\dfrac{10}{50}=0{,}2<1$ : système réducteur.
\end{exemple}

\methode{Reconnaître le sens de rotation d'un système}
Engrenage : sens opposés. Poulies-courroie et chaîne-pignon : sens identiques.
\begin{exemple}
Sur un vélo, le pédalier et la roue arrière (reliés par une chaîne) tournent dans le même
sens.
\end{exemple}

\section{Sujets type BEPC}

\sujetbac[Engrenage réducteur]
\begin{enumerate}[label=\arabic*)]
\item Donne la relation entre $N_1$, $Z_1$, $N_2$ et $Z_2$ pour un engrenage.
\item Une roue menante de $Z_1=20$ dents tourne à $N_1=1\,000$ tr/min ; la roue menée a
$Z_2=40$ dents. Calcule $N_2$.
\item Ce système est-il réducteur ou multiplicateur de vitesse ? Justifie.
\item Quel est l'intérêt d'un système réducteur de vitesse dans une machine ?
\end{enumerate}

\sujetbac[Transmission par courroie]
\begin{enumerate}[label=\arabic*)]

$D_2=100$ mm. Calcule $N_2$.
\item Calcule le rapport de transmission $r$ de ce système.
\item Pourquoi une courroie peut-elle occasionnellement glisser sur une poulie ?
\end{enumerate}

\item Ce système est-il réducteur ou multiplicateur de vitesse ?
\item Pourquoi ce choix est-il adapté à un vélo (vitesse de la roue par rapport au
pédalage) ?
\end{enumerate}

\section{Corrigés}
\begin{multicols}{2}

\corrige{de l'exercice 1}
L'engrenage, le système poulies-courroie et le système chaîne-pignon.

\corrige{de l'exercice 2}
Opposé (sens inversé).

\corrige{de l'exercice 3}
Identique à celui de la poulie menante.

\corrige{de l'exercice 4}
La chaîne (denture engrenant sans glissement).

\corrige{de l'exercice 5}
$N_2=1\,000\times\dfrac{20}{40}=500$ tr/min.

\corrige{de l'exercice 6}
$N_2=3\,000\times\dfrac{10}{50}=600$ tr/min.

\corrige{de l'exercice 7}
$N_2=600\times\dfrac{30}{20}=900$ tr/min. $r=1{,}5>1$ : système multiplicateur.

\corrige{de l'exercice 8}
$N_2=1\,000\times\dfrac{50}{100}=500$ tr/min.

\corrige{de l'exercice 9}
$N_2=2\,000\times\dfrac{30}{60}=1\,000$ tr/min.

\corrige{de l'exercice 10}
$r=\dfrac{40}{80}=0{,}5<1$ : système réducteur.

\corrige{du sujet 1}
1) $N_1\times Z_1=N_2\times Z_2$.\\
2) $N_2=1\,000\times\dfrac{20}{40}=500$ tr/min.\\
3) Réducteur, car $N_2<N_1$ (ou $r=Z_1/Z_2=0{,}5<1$).\\
4) Il permet de diminuer la vitesse tout en augmentant le couple (la force) transmis,
utile pour déplacer des charges importantes.

\corrige{du sujet 2}
1) Le même sens.\\
2) $N_2=1\,000\times\dfrac{50}{100}=500$ tr/min.\\
3) $r=\dfrac{500}{1\,000}=0{,}5$.\\
4) Car la courroie n'est pas rigidement fixée aux poulies : un effort trop important ou
une tension insuffisante peut la faire patiner.

\corrige{du sujet 3}
1) Elle ne glisse pas, contrairement à la courroie (transmission plus fiable).\\
2) $r=\dfrac{40}{16}=2{,}5$.\\
3) Multiplicateur, car $r>1$.\\
4) Cela permet à la roue de tourner plus vite que le pédalier, donc au vélo d'avancer
plus vite pour un même effort de pédalage.

\end{multicols}
