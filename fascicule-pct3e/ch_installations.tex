\chapter{Installations électriques domestiques et maintenance}

\noindent\textit{Une installation électrique domestique doit avant tout protéger les
personnes et les biens. Ce dernier chapitre présente son organisation, les dispositifs de
sécurité qui la composent, et les bons réflexes de maintenance et de prévention.}

\section{Organisation d'une installation domestique}

\begin{definition}[Tableau électrique]
Le \textbf{tableau électrique} centralise la distribution de l'électricité dans un
logement. Il regroupe le \textbf{disjoncteur général} (coupe l'ensemble de
l'installation) et plusieurs \textbf{disjoncteurs divisionnaires}, chacun protégeant un
circuit particulier (éclairage, prises, plaque de cuisson\dots).
\end{definition}

\section{Les dispositifs de protection}

\begin{propriete}[Disjoncteur]
Un \textbf{disjoncteur} coupe automatiquement le circuit qu'il protège lorsqu'un
\textbf{courant trop important} le traverse (court-circuit ou surcharge). Contrairement
au fusible, il peut être \textbf{réarmé} après le défaut, sans être remplacé.
\end{propriete}

\begin{propriete}[Fusible]
Un \textbf{fusible} contient un fil métallique qui \textbf{fond} lorsque le courant
dépasse une valeur limite, coupant ainsi le circuit. Contrairement au disjoncteur, il doit
être \textbf{remplacé} après chaque déclenchement.
\end{propriete}

\begin{propriete}[Disjoncteur différentiel et mise à la terre]
Le \textbf{disjoncteur différentiel} protège les \textbf{personnes} : il compare
l'intensité qui entre et l'intensité qui sort d'un circuit ; si une différence apparaît
(fuite de courant, par exemple à travers le corps d'une personne), il coupe
immédiatement l'alimentation. La \textbf{mise à la terre} relie la carcasse métallique
des appareils au sol : en cas de défaut, le courant de fuite s'écoule vers la terre plutôt
que de traverser une personne qui toucherait l'appareil.
\end{propriete}

\begin{remarque}
Disjoncteur différentiel et mise à la terre fonctionnent \textbf{ensemble} : la mise à la
terre offre un chemin de fuite au courant, et le différentiel détecte cette fuite pour
couper le circuit.
\end{remarque}

\section{Calibre et puissance admissible}

\begin{definition}[Calibre d'un disjoncteur]
Le \textbf{calibre} (en ampères, A) d'un disjoncteur divisionnaire indique l'intensité
maximale qu'il laisse passer avant de couper le circuit. La \textbf{puissance maximale}
admissible sur ce circuit se calcule par :
\[
P_{\max}=U\times I_{\text{calibre}}.
\]
\end{definition}

\begin{exemple}
Un circuit protégé par un disjoncteur de calibre $I=16$ A, sous $U=220$ V :
$P_{\max}=220\times16=3\,520$ W. Brancher des appareils dont la puissance totale dépasse
cette valeur déclenche le disjoncteur.
\end{exemple}

\begin{remarque}
C'est pour cette raison que les circuits les plus gourmands en puissance (plaque de
cuisson, chauffe-eau) utilisent des disjoncteurs de calibre plus élevé (souvent $20$ ou
$32$ A) que les circuits d'éclairage (souvent $10$ A).
\end{remarque}

\section{Maintenance et prévention des risques}

\begin{propriete}[Bonnes pratiques]
Une installation électrique se maintient en sécurité en respectant quelques règles
simples : \textbf{couper le disjoncteur} avant toute intervention sur un circuit, ne
\textbf{jamais surcharger} une prise multiple, faire \textbf{vérifier périodiquement}
l'installation, et faire appel à un \textbf{professionnel qualifié} pour toute
modification importante.
\end{propriete}

\begin{remarque}
Un fil dénudé, une prise fissurée ou une odeur de brûlé sont des signes d'alerte qui
doivent conduire à couper immédiatement le circuit concerné.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Tableau électrique.} Disjoncteur général $+$ disjoncteurs divisionnaires (un par
circuit).\\[4pt]
\textbf{Disjoncteur.} Coupe le courant en cas de surintensité ; réarmable.
\textbf{Fusible.} Fond et doit être remplacé.\\[4pt]
\textbf{Différentiel $+$ terre.} Protègent les personnes contre les fuites de courant
(chemin de fuite $+$ détection).\\[4pt]
\textbf{Puissance admissible.} $P_{\max}=U\times I_{\text{calibre}}$.\\[4pt]
\textbf{Piège fréquent.} Le disjoncteur protège le circuit (contre les surintensités) ;
le différentiel protège les personnes (contre les fuites de courant) : ce sont deux rôles
complémentaires, pas identiques.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer la puissance maximale admissible sur un circuit}
Appliquer $P_{\max}=U\times I_{\text{calibre}}$.
\begin{exemple}
$U=220$ V, $I_{\text{calibre}}=20$ A : $P_{\max}=220\times20=4\,400$ W.
\end{exemple}

\methode{Vérifier si un ensemble d'appareils peut être branché sans risque}
Additionner les puissances des appareils, puis comparer à $P_{\max}$ du circuit.
\begin{exemple}
Sur un circuit de $P_{\max}=3\,520$ W, brancher un four ($2\,000$ W) et un grille-pain
($1\,000$ W) donne $3\,000$ W $<3\,520$ W : c'est possible sans déclenchement.
\end{exemple}

\methode{Distinguer les rôles des différents dispositifs de sécurité}
Se demander ce qui est protégé : le \textbf{circuit} (disjoncteur, fusible) ou la
\textbf{personne} (différentiel, mise à la terre).
\begin{exemple}
Un court-circuit déclenche le disjoncteur ; un défaut d'isolement dangereux pour une
personne déclenche le différentiel.
\end{exemple}
