\chapter{Production et transport de l'énergie électrique}

\noindent\textit{L'électricité que nous utilisons chaque jour est produite dans des
centrales, transportée sur de longues distances à très haute tension, puis abaissée pour
être livrée en toute sécurité jusque dans nos maisons. Ce chapitre explique comment et
pourquoi.}
\vspace{8pt}

\connecteBanner

\section{Produire de l'électricité : l'alternateur}

\begin{definition}[Alternateur]
Un \textbf{alternateur} est une machine qui transforme de l'\textbf{énergie mécanique}
(rotation) en \textbf{énergie électrique}. Un aimant (ou un électroaimant) tourne à
proximité d'une bobine : ce mouvement fait apparaître une tension aux bornes de la bobine,
c'est le phénomène d'\textbf{induction électromagnétique}.
\end{definition}

\begin{propriete}[Centrales électriques]
Selon la source d'énergie mécanique qui entraîne l'alternateur, on distingue plusieurs
types de centrales : \textbf{hydraulique} (chute d'eau, barrage), \textbf{thermique}
(combustion de charbon, gaz ou fioul faisant tourner une turbine à vapeur), \textbf{solaire}
(panneaux photovoltaïques, sans alternateur) ou \textbf{éolienne} (vent faisant tourner des
pales reliées à un alternateur).
\end{propriete}

\section{Pourquoi transporter à haute tension ?}

\begin{propriete}[Pertes en ligne]
Une ligne électrique de résistance $R$ traversée par un courant $I$ dissipe une puissance
sous forme de chaleur :
\[
P_{\text{pertes}}=R\times I^2.
\]
Pour transporter une puissance $P$ donnée sous une tension $U$, l'intensité vaut
$I=\dfrac PU$. \textbf{Plus $U$ est grande, plus $I$ est petite}, donc plus les pertes
$R\times I^2$ sont faibles.
\end{propriete}

\begin{exemple}
Une ligne de résistance $R=5\ \Omega$ transporte une puissance $P=50\,000$ W. Sous
$U=1\,000$ V : $I=\dfrac{50\,000}{1\,000}=50$ A et $P_{\text{pertes}}=5\times50^2=12\,500$
W (25\,\% de la puissance transportée est perdue !).
\end{exemple}

\begin{remarque}
C'est pour cette raison que l'électricité est transportée sur de longues distances sous
des tensions très élevées (parfois plusieurs centaines de milliers de volts), bien
supérieures à la tension d'utilisation domestique.
\end{remarque}

\section{Le transformateur}

\begin{definition}[Transformateur]
Un \textbf{transformateur} est un appareil qui modifie une tension alternative sans
changer sa fréquence. Il comporte deux bobines (primaire, $n_1$ spires ; secondaire, $n_2$
spires) enroulées sur un même noyau de fer. Le rapport des tensions est égal au rapport
des nombres de spires :
\[
\frac{U_2}{U_1}=\frac{n_2}{n_1}.
\]
Un transformateur est \textbf{élévateur} si $n_2>n_1$ (il augmente la tension) et
\textbf{abaisseur} si $n_2<n_1$ (il diminue la tension).
\end{definition}

\begin{propriete}[Conservation de la puissance (transformateur idéal)]
En négligeant les pertes, la puissance électrique se conserve entre le primaire et le
secondaire :
\[
U_1\times I_1=U_2\times I_2.
\]
\end{propriete}

\begin{exemple}
Un transformateur élévateur a $n_1=100$ spires et $n_2=2\,000$ spires. Il reçoit
$U_1=220$ V : $U_2=220\times\dfrac{2\,000}{100}=4\,400$ V. Si $I_1=50$ A, alors
$I_2=\dfrac{U_1\times I_1}{U_2}=\dfrac{220\times50}{4\,400}=2{,}5$ A.
\end{exemple}

\section{De la centrale à la maison}

\begin{synthese}
La chaîne de l'énergie électrique suit quatre étapes : \textbf{production} (alternateur en
centrale, tension de quelques milliers de volts) $\rightarrow$ \textbf{élévation} (transformateur
élévateur, très haute tension, pour limiter les pertes en ligne) $\rightarrow$
\textbf{transport} (lignes à haute tension sur de longues distances) $\rightarrow$
\textbf{abaissement} (transformateurs abaisseurs successifs, jusqu'à la tension
domestique, $220$ V au Cameroun).
\end{synthese}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Alternateur.} Transforme l'énergie mécanique en énergie électrique par
induction.\\[4pt]
\textbf{Pertes en ligne.} $P_{\text{pertes}}=R\times I^2$ : on transporte à haute tension
pour réduire $I$ et donc les pertes.\\[4pt]
\textbf{Transformateur.} $\dfrac{U_2}{U_1}=\dfrac{n_2}{n_1}$ ; conservation de puissance
(idéale) $U_1I_1=U_2I_2$.\\[4pt]
\textbf{Chaîne complète.} Production $\to$ élévation $\to$ transport $\to$ abaissement
$\to$ distribution domestique.\\[4pt]
\textbf{Piège fréquent.} Une tension élevée au transport ne veut pas dire une intensité
élevée : c'est l'inverse, on élève $U$ pour \textbf{diminuer} $I$.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer les pertes en ligne}
Calculer $I=\dfrac PU$, puis appliquer $P_{\text{pertes}}=R\times I^2$.
\begin{exemple}
$R=2\ \Omega$, $P=20\,000$ W, $U=2\,000$ V : $I=10$ A, $P_{\text{pertes}}=2\times100=200$
W.
\end{exemple}

\methode{Calculer la tension au secondaire d'un transformateur}
Appliquer $U_2=U_1\times\dfrac{n_2}{n_1}$.
\begin{exemple}
$n_1=500$, $n_2=100$, $U_1=15\,000$ V : $U_2=15\,000\times\dfrac{100}{500}=3\,000$ V.
\end{exemple}

\methode{Reconnaître un transformateur élévateur ou abaisseur}
Comparer $n_1$ et $n_2$ (ou $U_1$ et $U_2$) : si $n_2>n_1$, élévateur ; si $n_2<n_1$,
abaisseur.
\begin{exemple}
$n_1=2\,000$, $n_2=100$ : $n_2<n_1$, c'est un transformateur abaisseur.
\end{exemple}

\methode{Utiliser la conservation de puissance d'un transformateur idéal}
Appliquer $U_1I_1=U_2I_2$ pour calculer l'intensité manquante.
\begin{exemple}
$U_1=220$ V, $I_1=10$ A, $U_2=4\,400$ V : $I_2=\dfrac{220\times10}{4\,400}=0{,}5$ A.
\end{exemple}

1) $P_{\text{pertes}}=R\times I^2$.\\

4) Une tension élevée réduit l'intensité transportée pour une même puissance, donc réduit

1) $\dfrac{U_2}{U_1}=\dfrac{n_2}{n_1}$.\\

4) $I_2=\dfrac{220\times40}{4\,400}=2$ A.

1) Production, élévation, transport, abaissement/distribution.\\
2) Un transformateur élévateur, pour réduire l'intensité transportée et donc les pertes
en ligne.\\

4) L'énergie mécanique en énergie électrique.
