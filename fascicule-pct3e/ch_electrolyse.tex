\chapter{Électrolyse et synthèse de l'eau}

\noindent\textit{L'eau, molécule si familière, est en réalité composée de deux gaz :
l'hydrogène et l'oxygène. Ce chapitre montre comment décomposer l'eau en ces deux gaz
(électrolyse), et comment, à l'inverse, les recombiner pour reformer de l'eau (synthèse).}
\vspace{8pt}

\connecteBanner

\section{L'électrolyse de l'eau}

\begin{definition}[Électrolyse de l'eau]
L'\textbf{électrolyse de l'eau} est la décomposition de l'eau, sous l'effet d'un courant
électrique, en dihydrogène et dioxygène. On recueille le \textbf{dihydrogène} $\mathrm{H_2}$
à la \textbf{cathode} (électrode négative) et le \textbf{dioxygène} $\mathrm{O_2}$ à
l'\textbf{anode} (électrode positive).
\end{definition}

\begin{propriete}[Équation-bilan de l'électrolyse]
\[
2\mathrm{H_2O}\longrightarrow2\mathrm{H_2}+\mathrm{O_2}.
\]
\end{propriete}

\begin{propriete}[Rapport des volumes de gaz]
Le volume de dihydrogène recueilli est \textbf{le double} du volume de dioxygène recueilli
:
\[
V(\mathrm{H_2})=2\times V(\mathrm{O_2}).
\]
\end{propriete}

\end{illus}

\begin{exemple}
On recueille $50$ cm$^{3}$ de dihydrogène à la cathode. Le volume de dioxygène recueilli à
l'anode est $V(\mathrm{O_2})=\dfrac{50}{2}=25$ cm$^{3}$.
\end{exemple}

\begin{propriete}[Tests d'identification des gaz]
Le \textbf{dihydrogène} est identifié par le test à la flamme : approché d'une flamme, il
provoque une petite détonation (« aboiement »). Le \textbf{dioxygène} est identifié par le
test à la buchette incandescente : une buchette qui rougeoie sans flamme se
\textbf{ranime} (se rallume) au contact du dioxygène.
\end{propriete}

\section{La synthèse de l'eau}

\begin{definition}[Synthèse de l'eau]
La \textbf{synthèse de l'eau} est la réaction \textbf{inverse} de l'électrolyse : le
dihydrogène brûle dans le dioxygène pour former de l'eau.
\end{definition}

\begin{propriete}[Équation-bilan de la synthèse]
\[
2\mathrm{H_2}+\mathrm{O_2}\longrightarrow2\mathrm{H_2O}.
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[every node/.style={font=\small},align=center]
\node[draw=onbo,rounded corners=3pt,fill=onbsoft,minimum width=3.4cm,minimum height=1.3cm] (L) {$2\mathrm{H_2}+\mathrm{O_2}$\\[2pt]\footnotesize (combustion)};
\node[right=1cm of L,font=\Large] {$\longrightarrow$};
\node[draw=onbgreen,rounded corners=3pt,fill=onbgreen!10,minimum width=3.4cm,minimum height=1.3cm,right=4cm of L] (R) {$2\mathrm{H_2O}$\\[2pt]\footnotesize (eau)};
\end{tikzpicture}
\fig{Figure — La synthèse de l'eau : combustion du dihydrogène dans le dioxygène.}
\end{illus}

\begin{exemple}
Vérification de l'équilibrage : à gauche, $4$ atomes H et $2$ atomes O ; à droite ($2$
molécules $\mathrm{H_2O}$), $4$ atomes H et $2$ atomes O. L'équation est équilibrée.
\end{exemple}

\begin{remarque}
Électrolyse et synthèse de l'eau sont deux réactions \textbf{opposées} : l'électrolyse
consomme de l'énergie électrique pour décomposer l'eau, tandis que la synthèse
\textbf{libère} de l'énergie (sous forme de chaleur et de lumière) en la reformant.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Électrolyse de l'eau.} $2\mathrm{H_2O}\to2\mathrm{H_2}+\mathrm{O_2}$ ;
$\mathrm{H_2}$ à la cathode ($-$), $\mathrm{O_2}$ à l'anode ($+$).\\[4pt]
\textbf{Rapport des volumes.} $V(\mathrm{H_2})=2\times V(\mathrm{O_2})$.\\[4pt]
\textbf{Tests d'identification.} $\mathrm{H_2}$ : détonation à la flamme. $\mathrm{O_2}$ :
ranime une buchette incandescente.\\[4pt]
\textbf{Synthèse de l'eau.} $2\mathrm{H_2}+\mathrm{O_2}\to2\mathrm{H_2O}$ (réaction inverse
de l'électrolyse).\\[4pt]
\textbf{Piège fréquent.} Ne pas inverser cathode et anode : c'est à la \textbf{cathode}
(borne négative) que se dégage le \textbf{plus grand} volume de gaz ($\mathrm{H_2}$).
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Écrire l'équation-bilan de l'électrolyse de l'eau}
Partir de l'eau comme réactif, et du dihydrogène et du dioxygène comme produits, puis
équilibrer avec les coefficients $2$, $2$ et $1$.
\begin{exemple}
$2\mathrm{H_2O}\to2\mathrm{H_2}+\mathrm{O_2}$.
\end{exemple}

\methode{Calculer le volume de dioxygène connaissant le volume de dihydrogène}
Diviser par $2$ le volume de dihydrogène recueilli.
\begin{exemple}
$V(\mathrm{H_2})=40$ cm$^{3}$ : $V(\mathrm{O_2})=\dfrac{40}2=20$ cm$^{3}$.
\end{exemple}

\methode{Calculer le volume de dihydrogène connaissant le volume de dioxygène}
Multiplier par $2$ le volume de dioxygène recueilli.
\begin{exemple}
$V(\mathrm{O_2})=15$ cm$^{3}$ : $V(\mathrm{H_2})=2\times15=30$ cm$^{3}$.
\end{exemple}

\methode{Identifier un gaz recueilli lors de l'électrolyse}
Utiliser le test caractéristique : détonation à la flamme pour $\mathrm{H_2}$, buchette
incandescente qui se ranime pour $\mathrm{O_2}$.
\begin{exemple}
Un gaz ranime une buchette incandescente : c'est du dioxygène.
\end{exemple}

\methode{Écrire et vérifier l'équation-bilan de la synthèse de l'eau}
Partir du dihydrogène et du dioxygène comme réactifs, de l'eau comme produit, équilibrer,
puis vérifier en comptant les atomes de chaque côté.
\begin{exemple}
$2\mathrm{H_2}+\mathrm{O_2}\to2\mathrm{H_2O}$ : $4$ atomes H et $2$ atomes O de chaque
côté.
\end{exemple}

\section{Exercices d'application}
\begin{multicols}{2}

\rubrique{Électrolyse de l'eau}
\exo{}\diff{1} Écrire l'équation-bilan de l'électrolyse de l'eau.

\exo{}\diff{1} À quelle électrode recueille-t-on le dihydrogène ? le dioxygène ?

\exo{}\diff{2} Quel est le rapport entre le volume de dihydrogène et le volume de
dioxygène recueillis à l'électrolyse de l'eau ?

\exo{}\diff{2} Pourquoi dit-on que l'électrolyse est une décomposition ?

\rubrique{Calculs de volumes}
\exo{}\diff{1} On recueille $40$ cm$^{3}$ de dihydrogène à la cathode. Calculer le volume
de dioxygène recueilli à l'anode.

\exo{}\diff{2} On recueille $15$ cm$^{3}$ de dioxygène à l'anode. Calculer le volume de
dihydrogène recueilli à la cathode.

\exo{}\diff{2} On recueille $100$ cm$^{3}$ de dihydrogène à la cathode. Calculer le
volume de dioxygène recueilli à l'anode.

\exo{}\diff{3} Au cours d'une électrolyse de l'eau, on recueille un gaz sur la cathode de
volume $25$ cL. Calculer le volume de gaz qu'on recueille sur l'anode.

\rubrique{Tests d'identification}
\exo{}\diff{1} Quel test permet d'identifier le dihydrogène ?

\exo{}\diff{1} Quel test permet d'identifier le dioxygène ?

\exo{}\diff{2} Un gaz provoque une détonation au contact d'une flamme. De quel gaz
s'agit-il ?

\exo{}\diff{2} Une buchette incandescente se ranime au contact d'un gaz recueilli à
l'anode. De quel gaz s'agit-il ? Ce résultat est-il cohérent avec l'électrode où il a été
recueilli ?

\rubrique{Synthèse de l'eau}
\exo{}\diff{1} Écrire l'équation-bilan de la synthèse de l'eau.

\exo{}\diff{2} Quelle est la différence entre l'électrolyse et la synthèse de l'eau ?

\exo{}\diff{2} Vérifier que l'équation $2\mathrm{H_2}+\mathrm{O_2}\to2\mathrm{H_2O}$ est
équilibrée, en comptant les atomes de chaque côté.

\exo{}\diff{3} $4$ g de dihydrogène réagissent complètement avec $32$ g de dioxygène pour
former de l'eau. Calculer la masse d'eau obtenue (loi de Lavoisier).
