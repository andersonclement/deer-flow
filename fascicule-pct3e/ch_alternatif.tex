\chapter{Tension alternative et énergie électrique domestique}

\noindent\textit{La tension livrée par le secteur n'est pas constante : elle varie
périodiquement au cours du temps. Ce chapitre étudie cette tension alternative, puis
s'intéresse à l'énergie électrique consommée par les appareils domestiques et à son coût.}
\vspace{8pt}

\connecteBanner

\section{La tension alternative sinusoïdale}

\begin{definition}[Tension alternative]
Une \textbf{tension alternative} change continuellement de valeur et de sens au cours du
temps, contrairement à une tension continue (pile). La tension du secteur est
\textbf{sinusoïdale} : elle se répète identique à elle-même après une durée constante
appelée \textbf{période} $T$ (en secondes, s).
\end{definition}

\begin{propriete}[Fréquence]
La \textbf{fréquence} $f$ (en hertz, Hz) est le nombre de périodes par seconde :
\[
f=\frac1T.
\]
Au Cameroun, comme dans la majorité des pays, le secteur délivre une tension alternative
de fréquence $f=50$ Hz.
\end{propriete}

\begin{exemple}
Pour $f=50$ Hz : $T=\dfrac1{50}=0{,}02$ s $=20$ ms.
\end{exemple}

\begin{remarque}
La valeur affichée sur les appareils ($220$ V) n'est pas la tension maximale atteinte à
chaque instant, mais une valeur moyenne appelée \textbf{valeur efficace} : elle correspond
à la tension continue qui produirait le même effet (chauffage, éclairage) que la tension
alternative réelle.
\end{remarque}

\section{Énergie électrique consommée}

\begin{propriete}[Énergie électrique]
L'\textbf{énergie électrique} $E$ consommée par un appareil de puissance $P$ pendant une
durée $t$ vaut :
\[
E=P\times t.
\]
Si $P$ est en watts (W) et $t$ en heures (h), $E$ s'exprime en \textbf{wattheures} (Wh).
On utilise plus couramment le \textbf{kilowattheure} (kWh) : $1$ kWh $=1\,000$ Wh, énergie
consommée par un appareil de $1\,000$ W fonctionnant pendant $1$ h.
\end{propriete}

\begin{exemple}
Une ampoule de $P=100$ W fonctionne pendant $t=5$ h : $E=100\times5=500$ Wh $=0{,}5$ kWh.
\end{exemple}

\section{La facture d'électricité}

\begin{definition}[Compteur et facture]
Le \textbf{compteur électrique} mesure l'énergie totale consommée, en kWh. La
\textbf{facture} d'électricité correspond au produit de l'énergie consommée par le prix
unitaire du kWh (fixé par le fournisseur).
\end{definition}

\begin{exemple}
Au tarif de $79$ FCFA/kWh, une consommation de $E=10$ kWh coûte
$10\times79=790$ FCFA.
\end{exemple}

\begin{remarque}
Pour réduire sa facture, on peut réduire la puissance des appareils utilisés, leur temps
de fonctionnement, ou remplacer les appareils énergivores (ex. : ampoules à incandescence)
par des appareils plus économes (ex. : ampoules LED).
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Tension alternative.} Change périodiquement ; période $T$ (s), fréquence
$f=\dfrac1T$ (Hz). Secteur : $50$ Hz, $220$ V (valeur efficace).\\[4pt]
\textbf{Énergie électrique.} $E=P\times t$ (Wh si $P$ en W et $t$ en h ; $1$ kWh
$=1\,000$ Wh).\\[4pt]
\textbf{Facture.} Coût $=$ énergie consommée (kWh) $\times$ prix unitaire du kWh.\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre puissance ($P$, en W, ce que consomme un
appareil à chaque instant) et énergie ($E=Pt$, en Wh, ce qui est facturé).
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer une période ou une fréquence}
Utiliser $f=\dfrac1T$ ou $T=\dfrac1f$, avec $T$ en secondes et $f$ en hertz.
\begin{exemple}
$T=0{,}02$ s : $f=\dfrac1{0{,}02}=50$ Hz.
\end{exemple}

\methode{Calculer une énergie électrique}
Appliquer $E=P\times t$, en veillant à convertir les unités si nécessaire (min $\to$ h).
\begin{exemple}
$P=1\,500$ W, $t=2$ h : $E=1\,500\times2=3\,000$ Wh $=3$ kWh.
\end{exemple}

\methode{Calculer le coût d'une consommation électrique}
Calculer l'énergie $E$ en kWh, puis multiplier par le prix unitaire du kWh.
\begin{exemple}
$E=15$ kWh à $79$ FCFA/kWh : coût $=15\times79=1\,185$ FCFA.
\end{exemple}

\section{Sujets type BEPC}

\sujetbac[Tension du secteur]
\begin{enumerate}[label=\arabic*)]
\item Qu'est-ce qu'une tension alternative sinusoïdale ?
\item La tension du secteur a une fréquence $f=50$ Hz. Calcule sa période $T$.
\item Quelle est la valeur efficace de la tension du secteur au Cameroun ?
\item Que mesure un voltmètre branché sur une prise domestique ?
\end{enumerate}

\sujetbac[Consommation et facture]
\begin{enumerate}[label=\arabic*)]
\item Donne la relation entre l'énergie électrique $E$, la puissance $P$ et la durée $t$.
\item Un téléviseur de puissance $P=120$ W fonctionne $t=5$ h par jour. Calcule l'énergie
consommée par jour, en Wh puis en kWh.
\item Au tarif de $79$ FCFA/kWh, calcule le coût de cette consommation journalière.
\item Calcule le coût sur un mois de $30$ jours.
\end{enumerate}

\sujetbac[Comparer deux appareils]
\begin{enumerate}[label=\arabic*)]
\item Un ventilateur de $P=75$ W fonctionne $t=6$ h par jour. Calcule l'énergie consommée
par jour en kWh.
\item Un second ventilateur, plus puissant, consomme $E=0{,}9$ kWh pour la même durée.
Calcule sa puissance.
\item Lequel des deux ventilateurs est le plus économe en énergie ?
\item Propose une explication à la différence de consommation entre les deux appareils.
\end{enumerate}

\corrige{de l'exercice 10}
Car $E=Pt$ : à puissance $P$ fixée, $E$ augmente proportionnellement avec la durée $t$
d'utilisation, donc le coût (proportionnel à $E$) augmente aussi.

\corrige{du sujet 1}
1) Une tension qui varie périodiquement, en suivant une courbe sinusoïdale.\\
2) $T=\dfrac1{50}=0{,}02$ s.\\
3) $220$ V.\\
4) Sa valeur efficace ($220$ V).

1) $E=P\times t$.\\
2) $E=120\times5=600$ Wh $=0{,}6$ kWh.\\

1) $E=75\times6=450$ Wh $=0{,}45$ kWh.\\
2) $P=\dfrac{0{,}9\times1\,000}6=150$ W.\\
3) Le premier ventilateur ($75$ W, $0{,}45$ kWh/jour), plus économe.\\
4) Le second ventilateur a une puissance plus élevée (moteur plus puissant, plus de
brassage d'air), donc consomme davantage d'énergie pour la même durée.
