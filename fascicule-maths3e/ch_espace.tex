\chapter{Solides usuels : sections planes et volumes}

\noindent\textit{Réservoirs, silos, tentes, ballons : de nombreux objets qui nous entourent
ont la forme d'un cylindre, d'une pyramide, d'un cône ou d'une sphère. Ce chapitre reprend
les formules d'aire et de volume de ces solides usuels, et étudie ce qui se passe lorsqu'on
coupe une pyramide ou un cône par un plan parallèle à sa base.}
\vspace{8pt}

\connecteBanner

\section{Rappels : prisme droit et cylindre de révolution}

\begin{definition}[Prisme droit]
Un \textbf{prisme droit} est un solide formé de deux bases superposables et parallèles,
reliées par des faces latérales rectangulaires perpendiculaires aux bases.
\end{definition}

\begin{propriete}[Volume et aire latérale d'un prisme droit]
\[
V=\mathcal{A}_{\text{base}}\times h \qquad\qquad \mathcal{A}_{\text{lat}}=P_{\text{base}}\times h
\]
où $h$ est la hauteur du prisme et $P_{\text{base}}$ le périmètre de la base.
\end{propriete}

\begin{definition}[Cylindre de révolution]
Un \textbf{cylindre de révolution} de rayon $r$ et de hauteur $h$ est engendré par un
rectangle qui tourne autour d'un de ses côtés. C'est un cas particulier de prisme dont la
base est un disque.
\end{definition}

\begin{propriete}[Cylindre de révolution]
\[
\mathcal{A}_{\text{lat}}=2\pi r h \qquad\qquad V=\pi r^{2}h
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[scale=0.85]
\draw[onbline,line width=0.8pt] (1.6,0) arc (0:-180:1.6 and 0.5);
\draw[onbline,dashed,line width=0.6pt] (1.6,0) arc (0:180:1.6 and 0.5);
\draw[onbline,line width=1pt] (-1.6,0) -- (-1.6,3);
\draw[onbline,line width=1pt] (1.6,0) -- (1.6,3);
\draw[onbo,line width=1.2pt,fill=onbsoft] (0,3) ellipse (1.6 and 0.5);
\draw[onbo,line width=1pt] (0,3) -- (1.6,3);
\node[onbo,above] at (0.8,3) {\small $r$};
\draw[onbmuted,line width=0.6pt] (2.1,0)--(2.3,0)--(2.3,3)--(2.1,3);
\node[onbmuted,right] at (2.3,1.5) {\small $h$};
\end{tikzpicture}
\fig{Figure — Cylindre de révolution : rayon $r$, hauteur $h$.}
\end{illus}

\begin{exemple}
Cylindre de rayon $r=5$ cm et de hauteur $h=9$ cm : $\mathcal{A}_{\text{lat}}=2\pi\times5\times9=90\pi$ cm$^{2}$
et $V=\pi\times5^{2}\times9=225\pi$ cm$^{3}$.
\end{exemple}

\begin{remarque}
On rappelle les conversions de volume utiles : $1\ \text{dm}^{3}=1$ L et
$1\ \text{m}^{3}=1\,000$ L. Pour convertir un volume en litres, on l'exprime d'abord en
dm$^{3}$.
\end{remarque}

\section{Pyramide}

\begin{definition}[Pyramide]
Une \textbf{pyramide} est un solide formé d'une base polygonale et de faces latérales
triangulaires qui se rejoignent en un même point, le \textbf{sommet}. La \textbf{hauteur}
est la distance du sommet au plan de la base. Une pyramide est \textbf{régulière} si sa base
est un polygone régulier et si le sommet se projette au centre de la base ; l'\textbf{apothème}
est alors la hauteur d'une face latérale (distance du sommet au milieu d'un côté de la base).
\end{definition}

\begin{propriete}[Volume d'une pyramide]
Pour toute pyramide, de base d'aire $\mathcal{A}_{\text{base}}$ et de hauteur $h$ :
\[
V=\frac{1}{3}\times\mathcal{A}_{\text{base}}\times h
\]
\end{propriete}

\begin{propriete}[Aire latérale d'une pyramide régulière]
Si la pyramide est \textbf{régulière}, d'apothème $a$ et de base de périmètre $P_{\text{base}}$ :
\[
\mathcal{A}_{\text{lat}}=\frac{1}{2}\times P_{\text{base}}\times a
\]
\end{propriete}

\end{tikzpicture}

\end{illus}

\begin{exemple}
Pyramide régulière à base carrée de côté $c=6$ cm et de hauteur $h=4$ cm. L'apothème $a$ est
l'hypoténuse d'un triangle rectangle de côtés $h=4$ et $c/2=3$ (Pythagore) :
$a=\sqrt{4^{2}+3^{2}}=\sqrt{25}=5$ cm.
Aire latérale : $\mathcal{A}_{\text{lat}}=\frac{1}{2}\times(4\times6)\times5=60$ cm$^{2}$.
Volume : $V=\frac{1}{3}\times6^{2}\times4=\frac{1}{3}\times144=48$ cm$^{3}$.
\end{exemple}

\begin{remarque}
La formule du volume $V=\frac13\mathcal{A}_{\text{base}}h$ vaut pour \textbf{toute} pyramide
(base quelconque, régulière ou non). En revanche, la formule de l'aire latérale avec
l'apothème ne s'applique qu'aux pyramides \textbf{régulières}.
\end{remarque}

\section{Cône de révolution}

\begin{definition}[Cône de révolution]
Un \textbf{cône de révolution} de rayon $r$ et de hauteur $h$ est engendré par un triangle
rectangle qui tourne autour d'un de ses côtés de l'angle droit. Le segment reliant le sommet
à un point du cercle de base s'appelle la \textbf{génératrice}, notée $l$.
\end{definition}

\begin{propriete}[Génératrice d'un cône]
Le rayon, la hauteur et la génératrice forment un triangle rectangle en le centre de la
base : d'après le théorème de Pythagore,
\[
l^{2}=r^{2}+h^{2}
\]
\end{propriete}

\begin{propriete}[Aires et volume d'un cône]
\[
\mathcal{A}_{\text{lat}}=\pi r l \qquad\quad \mathcal{A}_{\text{tot}}=\pi r^{2}+\pi rl=\pi r(r+l) \qquad\quad V=\frac{1}{3}\pi r^{2}h
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[scale=0.85]
\coordinate (O) at (0,0);
\coordinate (R) at (1.7,0);
\coordinate (Lp) at (-1.7,0);
\coordinate (S) at (0,3.2);
\draw[onbline,line width=0.8pt] (1.7,0) arc (0:-180:1.7 and 0.5);
\draw[onbline,dashed,line width=0.6pt] (1.7,0) arc (0:180:1.7 and 0.5);
\draw[onbo,line width=1.2pt] (S) -- (R);
\draw[onbo,line width=1.2pt] (S) -- (Lp);

\node[onbmuted,left] at (0,1.6) {\small $h$};
\node[onbo,above right] at (0.85,1.6) {\small $l$};
\end{tikzpicture}
\fig{Figure — Cône de révolution : le triangle $OSR$ est rectangle en $O$, $l^{2}=r^{2}+h^{2}$.}
\end{illus}

\begin{exemple}
Cône de rayon $r=9$ cm et de hauteur $h=12$ cm.
Génératrice : $l=\sqrt{9^{2}+12^{2}}=\sqrt{81+144}=\sqrt{225}=15$ cm.
Aire latérale : $\mathcal{A}_{\text{lat}}=\pi\times9\times15=135\pi$ cm$^{2}$.
Aire totale : $\mathcal{A}_{\text{tot}}=\pi\times9\times(9+15)=216\pi$ cm$^{2}$.
Volume : $V=\frac{1}{3}\pi\times9^{2}\times12=324\pi$ cm$^{3}$.
\end{exemple}

\begin{remarque}
Comme pour la pyramide, le volume d'un cône vaut le \textbf{tiers} du produit de l'aire de
la base par la hauteur : $V=\frac13\mathcal{A}_{\text{base}}h$ avec $\mathcal{A}_{\text{base}}=\pi r^{2}$.
\end{remarque}

\section{Sphère et boule}

\begin{definition}[Sphère et boule]
La \textbf{sphère} de centre $O$ et de rayon $r$ est l'ensemble des points de l'espace
situés à la distance $r$ de $O$ : c'est une \textbf{surface}. La \textbf{boule} de centre
$O$ et de rayon $r$ est le solide plein délimité par cette sphère.
\end{definition}

\begin{propriete}[Aire et volume de la sphère et de la boule]
\[
\mathcal{A}_{\text{sphère}}=4\pi r^{2} \qquad\qquad V_{\text{boule}}=\frac{4}{3}\pi r^{3}
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[scale=0.85]
\coordinate (O) at (0,0);
\draw[onbline,line width=1pt] (O) circle (1.6);
\draw[onbmuted,line width=0.6pt] (1.6,0) arc (0:-180:1.6 and 0.45);
\draw[onbmuted,dashed,line width=0.6pt] (1.6,0) arc (0:180:1.6 and 0.45);
\draw[onbo,line width=1.1pt] (O) -- (1.6,0);
\filldraw[onbo] (O) circle(1.2pt);
\node[onbo,below] at (0.8,0) {\small $r$};
\node[above left] at (O) {\small $O$};
\end{tikzpicture}
\fig{Figure — Sphère de centre $O$ et de rayon $r$.}
\end{illus}

\begin{exemple}
Sphère de rayon $r=6$ cm : $\mathcal{A}=4\pi\times6^{2}=144\pi$ cm$^{2}$ et
$V=\frac{4}{3}\pi\times6^{3}=\frac{4}{3}\pi\times216=288\pi$ cm$^{3}$.
\end{exemple}

\begin{remarque}
Pour une boule, on utilise parfois $\pi\approx3{,}14$ pour donner une valeur approchée du
volume, par exemple pour estimer une contenance en litres.
\end{remarque}

\section{Section par un plan parallèle à la base}

\begin{propriete}[Section d'une pyramide ou d'un cône]
Lorsqu'on coupe une pyramide (ou un cône) par un plan parallèle à sa base, à une distance du
sommet égale à un coefficient $k$ de la hauteur totale ($0<k<1$), le petit solide obtenu au
sommet est une \textbf{réduction} du solide entier, de coefficient $k$ :
\[
\text{longueurs}\times k \qquad\qquad \text{aires}\times k^{2} \qquad\qquad \text{volumes}\times k^{3}
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[scale=0.68]
\coordinate (S) at (0,4.2);
\coordinate (Ob) at (0,0);
\coordinate (Oc) at (0,1.4);
\draw[onbline,line width=0.8pt] (2.0,0) arc (0:-180:2.0 and 0.55);
\draw[onbline,dashed,line width=0.6pt] (2.0,0) arc (0:180:2.0 and 0.55);
\draw[onbline,line width=0.9pt] (Ob) -- (2.0,0);
\draw[onbline,line width=1.1pt] (S) -- (2.0,0);
\draw[onbline,line width=1.1pt] (S) -- (-2.0,0);
\draw[onbo,line width=1pt,fill=onbsoft] (1.333,1.4) arc (0:360:1.333 and 0.37);
\draw[onbo,line width=0.7pt] (Oc) -- (1.333,1.4);
\node[onbo,above] at (0.67,1.5) {\small $r$};
\node[below] at (1.0,0) {\small $R$};
\draw[onbmuted,line width=0.6pt] (-2.15,0)--(-2.35,0)--(-2.35,4.2)--(-2.15,4.2);
\node[onbmuted,left] at (-2.35,2.1) {\small $H$};
\draw[onbmuted,line width=0.6pt] (2.15,1.4)--(2.35,1.4)--(2.35,4.2)--(2.15,4.2);
\node[onbmuted,right] at (2.35,2.8) {\small $h'$};
\end{tikzpicture}
\fig{Figure — Section d'un cône par un plan parallèle à la base : $k=h'/H$, rayon réduit $r=kR$.}
\end{illus}

\begin{exemple}
Pyramide régulière de hauteur totale $H=12$ cm et de base carrée de côté $9$ cm. On la coupe
par un plan parallèle à la base, à $h'=8$ cm du sommet : $k=\dfrac{8}{12}=\dfrac{2}{3}$.
Côté réduit : $9\times\dfrac23=6$ cm. Aire réduite : $6^{2}=36$ cm$^{2}$.
Volume total : $V=\frac13\times9^{2}\times12=324$ cm$^{3}$.
Volume du petit solide réduit : $V\times\left(\dfrac23\right)^{3}=324\times\dfrac{8}{27}=96$ cm$^{3}$.
\end{exemple}

\begin{remarque}
Le coefficient $k$ se calcule toujours en comparant la \textbf{hauteur du petit solide}
(sommet au plan de coupe) à la \textbf{hauteur totale} du solide entier, jamais la distance
du plan de coupe à la base.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Prisme droit / cylindre.} $V=\mathcal{A}_{\text{base}}\times h$ ;
$\mathcal{A}_{\text{lat}}=P_{\text{base}}\times h$ (cylindre : $\mathcal{A}_{\text{lat}}=2\pi rh$,
$V=\pi r^{2}h$).\\[4pt]
\textbf{Pyramide.} $V=\frac13\mathcal{A}_{\text{base}}\times h$ (toute pyramide) ;
si régulière, $\mathcal{A}_{\text{lat}}=\frac12P_{\text{base}}\times a$ ($a$ = apothème).\\[4pt]
\textbf{Cône de révolution.} $l^{2}=r^{2}+h^{2}$ ; $\mathcal{A}_{\text{lat}}=\pi rl$ ;
$\mathcal{A}_{\text{tot}}=\pi r(r+l)$ ; $V=\frac13\pi r^{2}h$.\\[4pt]
\textbf{Sphère / boule.} $\mathcal{A}=4\pi r^{2}$ ; $V=\frac43\pi r^{3}$.\\[4pt]
\textbf{Section parallèle à la base (coefficient $k$).} Longueurs $\times k$, aires
$\times k^{2}$, volumes $\times k^{3}$.\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre l'aire \textbf{latérale} (faces courbes ou
triangulaires seulement) et l'aire \textbf{totale} (avec la ou les bases).
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer l'aire latérale et le volume d'un cylindre}
Utiliser $\mathcal{A}_{\text{lat}}=2\pi rh$ et $V=\pi r^{2}h$ directement à partir du rayon
et de la hauteur.
\begin{exemple}
$r=3$ cm, $h=7$ cm : $\mathcal{A}_{\text{lat}}=2\pi\times3\times7=42\pi$ cm$^{2}$,
$V=\pi\times3^{2}\times7=63\pi$ cm$^{3}$.
\end{exemple}

\methode{Calculer le volume d'une pyramide}
Calculer l'aire de la base, puis appliquer $V=\frac13\mathcal{A}_{\text{base}}\times h$.
\begin{exemple}
Pyramide à base carrée de côté $10$ cm, hauteur $12$ cm :
$V=\frac13\times10^{2}\times12=\frac13\times1\,200=400$ cm$^{3}$.
\end{exemple}

\methode{Calculer l'aire latérale d'une pyramide régulière}
Calculer l'apothème avec Pythagore (hauteur et demi-côté de la base), puis appliquer
$\mathcal{A}_{\text{lat}}=\frac12P_{\text{base}}\times a$.
\begin{exemple}
Base carrée de côté $10$ cm, hauteur $12$ cm : $a=\sqrt{12^{2}+5^{2}}=\sqrt{169}=13$ cm ;
$\mathcal{A}_{\text{lat}}=\frac12\times40\times13=260$ cm$^{2}$.
\end{exemple}

\methode{Calculer la génératrice et les aires d'un cône}
Calculer $l$ avec $l^{2}=r^{2}+h^{2}$, puis appliquer $\mathcal{A}_{\text{lat}}=\pi rl$ et
$\mathcal{A}_{\text{tot}}=\pi r(r+l)$.
\begin{exemple}
$r=8$ cm, $h=6$ cm : $l=\sqrt{64+36}=\sqrt{100}=10$ cm ; $\mathcal{A}_{\text{lat}}=80\pi$ cm$^{2}$,
$\mathcal{A}_{\text{tot}}=\pi\times8\times18=144\pi$ cm$^{2}$.
\end{exemple}

\methode{Calculer l'aire et le volume d'une sphère ou d'une boule}
Appliquer directement $\mathcal{A}=4\pi r^{2}$ pour la surface et $V=\frac43\pi r^{3}$ pour
le volume plein.
\begin{exemple}
$r=9$ cm : $\mathcal{A}=4\pi\times81=324\pi$ cm$^{2}$, $V=\frac43\pi\times729=972\pi$ cm$^{3}$.
\end{exemple}

\methode{Étudier une section parallèle à la base}
Calculer $k$ = (hauteur du petit solide) $\div$ (hauteur totale), puis multiplier les
longueurs par $k$, les aires par $k^{2}$ et les volumes par $k^{3}$.
\begin{exemple}
Cône de hauteur totale $18$ cm, rayon $6$ cm, coupé à $12$ cm du sommet : $k=\frac{12}{18}=\frac23$.
Rayon réduit $=6\times\frac23=4$ cm. Volume total $V=\frac13\pi\times36\times18=216\pi$ cm$^{3}$ ;
volume réduit $=216\pi\times\left(\frac23\right)^{3}=216\pi\times\frac{8}{27}=64\pi$ cm$^{3}$.
\end{exemple}

\methode{Résoudre un problème concret (réservoir, emballage)}
Identifier la forme du solide, relever les dimensions utiles dans l'énoncé, appliquer la
formule adaptée, puis convertir le résultat dans l'unité demandée (L, m$^{3}$...).
\begin{exemple}
Réservoir cylindrique de rayon $1{,}5$ m et de hauteur $4$ m :
$V=\pi\times1{,}5^{2}\times4\approx28{,}27$ m$^{3}$, soit environ $28\,274$ L.
\end{exemple}
