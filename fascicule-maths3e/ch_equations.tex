\chapter{Équations et inéquations du premier degré}

\noindent\textit{Résoudre une équation, c'est trouver la (ou les) valeur(s) de l'inconnue qui
rendent une égalité vraie. Une inéquation, elle, décrit un \textbf{ensemble} de valeurs
vérifiant une inégalité. Ces deux outils permettent de traduire et résoudre de nombreux
problèmes concrets — c'est là que le calcul littéral du chapitre précédent devient vraiment
utile.}
\vspace{8pt}

\section{Résoudre une équation du premier degré}

\begin{definition}[Équation, solution]
Une \textbf{équation} d'inconnue $x$ est une égalité contenant $x$. \textbf{Résoudre}
l'équation, c'est trouver toutes les valeurs de $x$ (les \textbf{solutions}) qui rendent
cette égalité vraie.
\end{definition}

\begin{propriete}[Règles de résolution]
On ne change pas les solutions d'une équation en :
\begin{itemize}
\item ajoutant (ou soustrayant) le \textbf{même nombre} aux deux membres ;
\item multipliant (ou divisant) les deux membres par un \textbf{même nombre non nul}.
\end{itemize}
Pour résoudre $ax+b=cx+d$, on regroupe les termes en $x$ d'un côté, les nombres de l'autre.
\end{propriete}

\begin{illus}
\begin{tikzpicture}
\draw[onbo,line width=2pt] (0,1.5) -- (5,1.5);
\draw[fill=onbline,draw=onbline] (2.3,0) -- (2.7,0) -- (2.5,1.5) -- cycle;
\draw[onbmuted,line width=1pt] (0.5,1.5) -- (0.5,0.6);
\draw[onbmuted,line width=1pt] (4.5,1.5) -- (4.5,0.6);
\draw[fill=onbsoft,draw=onbo,line width=0.8pt] (-0.2,0.2) rectangle (1.2,0.6);
\draw[fill=onbsoft,draw=onbo,line width=0.8pt] (3.8,0.2) rectangle (5.2,0.6);
\node at (0.5,0.4) {\small $x+3$};
\node at (4.5,0.4) {\small $7$};
\end{tikzpicture}
\fig{Figure — Une équation est comme une balance en équilibre.}
\end{illus}

\begin{exemple}
Résoudre $3x+5=2x+12$ : on soustrait $2x$ des deux côtés, $x+5=12$ ; on soustrait $5$,
$x=7$. La solution est $7$.
\end{exemple}

\begin{remarque}
Avant de résoudre, il faut souvent \textbf{développer} les parenthèses (chapitre précédent) :
$4(x-2)=3x+1 \iff 4x-8=3x+1 \iff x=9$.
\end{remarque}

\section{Équation produit nul}

\begin{propriete}[Produit nul]
Un produit de facteurs est nul \textbf{si et seulement si} l'un au moins de ses facteurs est
nul :
\[
A\times B=0 \iff A=0 \ \text{ou}\ B=0.
\]
\end{propriete}

\begin{illus}
\begin{tikzpicture}[>=Stealth,every node/.style={font=\small}]
\node[draw=onbo,fill=onbsoft,rounded corners=3pt,minimum width=2.6cm,minimum height=0.7cm] (p) {$A\times B=0$};
\node[draw=onbblue,fill=white,rounded corners=3pt,minimum width=2cm,minimum height=0.7cm,below=12mm of p,xshift=-2.3cm] (b2) {$A=0$};
\node[draw=onbgreen,fill=white,rounded corners=3pt,minimum width=2cm,minimum height=0.7cm,below=12mm of p,xshift=2.3cm] (b1) {$B=0$};
\draw[->,onbmuted,line width=1pt] (p) -- (b2);
\draw[->,onbmuted,line width=1pt] (p) -- (b1);
\node[below=1mm of p] {\tiny ou};
\end{tikzpicture}
\fig{Figure — Une équation produit nul se sépare en deux équations simples, reliées par
« ou ».}
\end{illus}

\begin{exemple}
$(x-3)(x+5)=0 \iff x-3=0\ \text{ou}\ x+5=0 \iff x=3\ \text{ou}\ x=-5$. L'équation a deux
solutions : $3$ et $-5$.
\end{exemple}

\begin{remarque}
Une équation produit nul suppose que l'expression est déjà (ou peut être) \textbf{factorisée}.
Si elle ne l'est pas, on factorise d'abord (chapitre précédent).
\end{remarque}

\section{Inéquations du premier degré}

\begin{propriete}[Règles de résolution d'une inéquation]
On ne change pas le sens d'une inégalité en ajoutant/soustrayant un même nombre, ni en
multipliant/divisant par un nombre \textbf{positif}. En revanche, multiplier ou diviser par
un nombre \textbf{négatif} \textbf{inverse le sens} de l'inégalité.
\end{propriete}

\begin{exemple}
$3x+1>7 \iff 3x>6 \iff x>2$ (on divise par $3>0$, le sens ne change pas). \quad
$-2x+5<1 \iff -2x<-4 \iff x>2$ (on divise par $-2<0$, le sens \textbf{change}).
\end{exemple}

\begin{illus}
\begin{tikzpicture}[>=Stealth]
\draw[onbline,line width=1pt] (-1,0) -- (6,0);
\foreach \x in {-1,...,6} \draw[onbmuted] (\x,0.06) -- (\x,-0.06);
\node[below] at (2,-0.15) {\small $2$};
\draw[onbo,line width=2pt,->] (2,0) -- (6,0);
\draw[onbo,fill=white,line width=1.2pt] (2,0) circle (2.5pt);
\node[above,onbo] at (2,0.25) {\small $x>2$};
\end{tikzpicture}
\fig{Figure — Solution $x>2$ sur une droite graduée : rond ouvert en $2$ (non solution).}
\end{illus}

\begin{remarque}
Si l'inégalité est large ($\ge$ ou $\le$), on utilise un rond \textbf{plein} sur la droite
graduée : le nombre lui-même fait partie de la solution.
\end{remarque}

\section{Mettre un problème en équation ou en inéquation}

\begin{propriete}[Méthode]
Choisir une inconnue $x$ (en précisant ce qu'elle représente), traduire chaque phrase de
l'énoncé par une expression ou une (in)égalité, résoudre, puis \textbf{revenir au contexte}
pour formuler la réponse (avec l'unité, et en vérifiant que la solution a un sens concret).
\end{propriete}

\draw[fill=onbgreen!25,draw=onbgreen,line width=0.9pt] (4.2,0) rectangle (9.45,0.7);

\begin{exemple}
Un nombre augmenté de $7$ donne le triple de ce nombre diminué de $3$ : $x+7=3x-3$, d'où
$10=2x$, soit $x=5$.
\end{exemple}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Équation $ax+b=cx+d$.} Regrouper les $x$ d'un côté, les nombres de l'autre ;
développer d'abord si besoin.\\[4pt]
\textbf{Produit nul.} $A\times B=0\iff A=0$ ou $B=0$ (factoriser si nécessaire avant).\\[4pt]
\textbf{Inéquation.} Mêmes règles qu'une équation, \textbf{sauf} : multiplier/diviser par un
nombre \textbf{négatif} inverse le sens de l'inégalité.\\[4pt]
\textbf{Droite graduée.} Rond \textbf{ouvert} pour $<,>$ (le nombre exclu) ; rond
\textbf{plein} pour $\le,\ge$ (le nombre inclus).\\[4pt]
\textbf{Mise en équation.} Choisir $x$, traduire l'énoncé, résoudre, puis conclure dans le
contexte (unité, vraisemblance de la réponse).\\[4pt]
\textbf{Piège fréquent.} Oublier d'inverser le sens d'une inéquation en divisant par un
nombre négatif ; oublier une des deux solutions d'une équation produit nul.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Résoudre une équation $ax+b=cx+d$}
Regrouper les termes en $x$ d'un côté et les nombres de l'autre, puis diviser par le
coefficient de $x$.
\begin{exemple}
$5x-7=2x+8 \iff 3x=15 \iff x=5$.
\end{exemple}

\methode{Résoudre une équation avec parenthèses}
Développer d'abord chaque membre, réduire, puis résoudre comme une équation classique.
\begin{exemple}
$2(x+3)-5=x+4 \iff 2x+1=x+4 \iff x=3$.
\end{exemple}

\methode{Résoudre une équation produit nul}
Vérifier que l'expression est factorisée (sinon, la factoriser), puis annuler chaque facteur
séparément.
\begin{exemple}
$(2x-1)(x+4)=0 \iff x=\dfrac12\ \text{ou}\ x=-4$.
\end{exemple}

\methode{Factoriser puis résoudre une équation produit nul}
Si l'équation n'est pas déjà sous forme de produit, la factoriser (facteur commun ou identité
remarquable) avant d'appliquer la règle du produit nul.
\begin{exemple}
$(x+2)^{2}-(x+2)(3x-1)=0$ : facteur commun $(x+2)$, donne
$(x+2)\big[(x+2)-(3x-1)\big]=(x+2)(3-2x)=0$, donc $x=-2$ ou $x=\dfrac32$.
\end{exemple}

\methode{Résoudre une inéquation}
Isoler $x$ comme pour une équation, en pensant à inverser le sens si l'on divise/multiplie
par un nombre négatif.
\begin{exemple}
$4-3x\ge-2 \iff -3x\ge-6 \iff x\le2$ (division par $-3<0$ : le sens change).
\end{exemple}

\methode{Représenter une inéquation sur une droite graduée}
Placer le nombre trouvé, choisir un rond ouvert ou plein selon l'inégalité, puis tracer une
flèche dans la bonne direction.
\begin{exemple}
$x\le2$ : rond \textbf{plein} en $2$, flèche vers la gauche (les nombres plus petits que
$2$).
\end{exemple}

\methode{Mettre un problème en équation ou en inéquation}
Choisir clairement l'inconnue, traduire chaque phrase, résoudre, puis répondre dans le
contexte du problème.
\begin{exemple}
Un rectangle a une longueur supérieure de $5$ cm à sa largeur $x$ ; son périmètre est $54$
cm : $2\big[x+(x+5)\big]=54 \iff 4x+10=54 \iff x=11$. Largeur $11$ cm, longueur $16$ cm.
\end{exemple}
