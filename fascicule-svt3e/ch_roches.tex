\chapter{Les roches sédimentaires}

\noindent\textit{Calcaire, grès, argile : les roches sédimentaires se forment à la
surface de la Terre, par un long processus d'usure, de transport et de dépôt. Ce
chapitre étudie leur formation, leurs différents types, et l'intérêt qu'elles
représentent pour reconstituer l'histoire de notre planète.}
\vspace{8pt}

\connecteBanner

\section{La formation des roches sédimentaires}

\begin{definition}[Roche sédimentaire]
Une \textbf{roche sédimentaire} est une roche formée en surface, par
\textbf{accumulation} de \textbf{sédiments} (débris de roches ou restes
d'organismes), organisés en couches successives appelées \textbf{strates}.
\end{definition}

\begin{propriete}[Quatre étapes de formation]
\begin{itemize}
\item \textbf{altération} : les roches en surface se fragmentent progressivement,
sous l'effet de l'eau, du vent et des variations de température ;
\item \textbf{transport} : les fragments obtenus (les sédiments) sont déplacés par
l'eau, le vent ou la glace ;
\item \textbf{sédimentation} : les sédiments se déposent, en couches successives,
dans un bassin (mer, lac, delta\dots) ;
\item \textbf{diagenèse} : sous le poids des couches qui s'accumulent au-dessus,
les sédiments sont \textbf{compactés} (l'eau est expulsée) puis \textbf{cimentés}
(des minéraux précipitent entre les grains), ce qui les transforme en roche solide.
\end{itemize}
\end{propriete}

\begin{exemple}
Un grain de sable, issu de l'altération d'un granite, peut être transporté par un
fleuve jusqu'à la mer, s'y déposer avec d'autres grains, puis, des milliers d'années
plus tard, être compacté et cimenté en une roche : le grès.
\end{exemple}

\section{Les types de roches sédimentaires}

\begin{propriete}[Deux grandes origines]
\begin{itemize}
\item les roches \textbf{détritiques}, formées de débris de roches préexistantes
(exemples : le grès, l'argile) ;
\item les roches \textbf{biogènes} (ou organiques), formées à partir de restes
d'organismes vivants (exemples : le calcaire coquillier, la craie, le charbon).
\end{itemize}
\end{propriete}

\begin{definition}[Principe de superposition]
Dans une série de strates non perturbée, la couche située le plus \textbf{bas} est
la plus \textbf{ancienne}, et la couche la plus \textbf{haute} est la plus
\textbf{récente} : c'est le \textbf{principe de superposition}.
\end{definition}

\begin{exemple}
Le calcaire coquillier, formé d'une accumulation de coquilles d'organismes marins, est
une roche sédimentaire biogène.
\end{exemple}

\section{Les intérêts des roches sédimentaires}

\begin{propriete}[Reconstituer les milieux anciens]
Les strates sédimentaires renferment souvent des \textbf{fossiles}, restes ou traces
d'êtres vivants du passé. En étudiant les fossiles et le type de sédiment d'une
strate, on peut reconstituer le \textbf{milieu} (marin, désertique, forestier\dots)
et parfois le \textbf{climat} qui existaient au moment du dépôt.
\end{propriete}

\begin{exemple}
La découverte de fossiles marins dans une strate, même loin de toute mer actuelle,
prouve que cette région était recouverte par la mer à l'époque du dépôt de cette
strate.
\end{exemple}

\begin{propriete}[Des usages nombreux]
\begin{itemize}
\item \textbf{matériaux de construction} : calcaire et grès taillés en pierres, ou
utilisés dans le ciment ;
\item \textbf{argile} : fabrication de briques et de poteries ;
\item \textbf{craie} : fabrication de peintures, de dentifrice ;
\item \textbf{charbon} : ressource énergétique, issue de la transformation
d'anciennes forêts enfouies.
\end{itemize}
\end{propriete}

\begin{remarque}
Piège fréquent : toutes les roches sédimentaires ne se forment pas de la même façon
(débris ou organismes), mais elles partagent toutes le même processus en quatre
étapes : altération, transport, sédimentation, diagenèse.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Roche sédimentaire.} Formée par accumulation de sédiments en strates.\\[4pt]
\textbf{Quatre étapes.} Altération $\to$ transport $\to$ sédimentation $\to$
diagenèse (compaction + cimentation).\\[4pt]
\textbf{Deux origines.} Détritique (débris de roches) ou biogène (restes
d'organismes).\\[4pt]
\textbf{Principe de superposition.} La strate la plus basse est la plus
ancienne.\\[4pt]
\textbf{Intérêts.} Reconstitution des milieux anciens grâce aux fossiles ; usages
variés (construction, briques, énergie\dots).\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre l'origine du sédiment (détritique ou
biogène) avec le processus de formation, toujours en quatre étapes.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Retracer les étapes de formation d'une roche sédimentaire}
Suivre l'ordre : altération, transport, sédimentation, diagenèse.
\begin{exemple}
Un sédiment ne devient une roche solide qu'après la diagenèse (compaction et
cimentation), pas dès son dépôt.
\end{exemple}

\methode{Appliquer le principe de superposition}
Dans une série de strates non perturbée, comparer les positions : la strate la plus
basse est la plus ancienne.
\begin{exemple}
Dans une falaise présentant plusieurs strates, la strate visible tout en bas s'est
déposée en premier.
\end{exemple}

\methode{Utiliser un fossile pour reconstituer un milieu ancien}
Identifier le type d'organisme fossilisé (marin, terrestre\dots) pour en déduire le
milieu qui existait au moment du dépôt de la strate.
\begin{exemple}
Un fossile de coquillage marin dans une strate indique un ancien milieu marin à cet
endroit.
\end{exemple}
