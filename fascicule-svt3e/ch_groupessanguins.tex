\chapter{Les groupes sanguins et la transfusion}

\noindent\textit{Tout le monde n'a pas le même sang : il existe plusieurs groupes
sanguins, définis par la présence ou l'absence de certaines molécules à la surface des
globules rouges. Connaître son groupe sanguin est indispensable avant toute transfusion,
sous peine de réaction grave.}
\vspace{8pt}

\connecteBanner

\section{Le système ABO}

\begin{definition}[Antigènes et anticorps sanguins]
La surface des globules rouges peut porter des \textbf{antigènes} A et/ou B. Le plasma
sanguin contient, en miroir, des \textbf{anticorps} dirigés contre les antigènes
\textbf{absents} des propres globules rouges de l'individu.
\end{definition}

\begin{propriete}[Les quatre groupes sanguins]
\begin{center}
\small
\begin{tabular}{lll}
\toprule
\textbf{Groupe} & \textbf{Antigènes (globules rouges)} & \textbf{Anticorps (plasma)} \\
\midrule
A & antigène A & anticorps anti-B \\
B & antigène B & anticorps anti-A \\
AB & antigènes A et B & aucun anticorps \\
O & aucun antigène & anticorps anti-A et anti-B \\
\bottomrule
\end{tabular}
\end{center}
\end{propriete}

\begin{remarque}
Un individu ne fabrique jamais d'anticorps contre ses propres antigènes : cela
provoquerait la destruction de ses propres globules rouges.
\end{remarque}

\section{La règle de compatibilité transfusionnelle}

\begin{propriete}[Le danger de l'agglutination]
Si l'on transfuse des globules rouges portant un antigène auquel le receveur possède
l'anticorps correspondant, les globules rouges transfusés s'\textbf{agglutinent}
(s'agglomèrent) : c'est une réaction dangereuse, pouvant boucher des vaisseaux sanguins.
\end{propriete}

\begin{propriete}[Règle de compatibilité]
Une transfusion est possible seulement si le \textbf{receveur ne possède pas
d'anticorps} contre les \textbf{antigènes du donneur}.
\end{propriete}

\begin{propriete}[Cas particuliers]
Le groupe \textbf{O} ne possède aucun antigène : il peut donner à \textbf{tous} les
groupes, c'est le \textbf{donneur universel}. Le groupe \textbf{AB} ne possède aucun
anticorps : il peut recevoir de \textbf{tous} les groupes, c'est le \textbf{receveur
universel}.
\end{propriete}

\section{Le facteur rhésus}

\begin{definition}[Facteur rhésus]
Le \textbf{facteur rhésus} est un antigène supplémentaire (antigène D), indépendant du
système ABO. Un individu est \textbf{Rh$+$} s'il porte cet antigène, \textbf{Rh$-$}
s'il ne le porte pas.
\end{definition}

\begin{propriete}[Règle du rhésus]
Un individu \textbf{Rh$-$} peut développer des anticorps anti-Rh s'il reçoit du sang
Rh$+$ : il ne doit donc recevoir que du sang \textbf{Rh$-$}. Un individu \textbf{Rh$+$}
peut recevoir indifféremment du sang Rh$+$ ou Rh$-$.
\end{propriete}

\begin{exemple}
Une personne de groupe O Rh$-$ est \textbf{donneur universel absolu} : son sang peut être
transfusé à n'importe quel individu, quel que soit son groupe ABO ou son rhésus.
\end{exemple}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Système ABO.} Groupe A (antigène A, anticorps anti-B) ; B (antigène B, anticorps
anti-A) ; AB (antigènes A et B, aucun anticorps) ; O (aucun antigène, anticorps anti-A et
anti-B).\\[4pt]
\textbf{Règle de compatibilité.} Le receveur ne doit pas posséder d'anticorps contre les
antigènes du donneur.\\[4pt]
\textbf{Cas particuliers.} O = donneur universel ; AB = receveur universel.\\[4pt]
\textbf{Rhésus.} Rh$-$ ne reçoit que du Rh$-$ ; Rh$+$ peut recevoir Rh$+$ ou Rh$-$.\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre antigènes (sur les globules rouges) et
anticorps (dans le plasma) : ce sont les anticorps du \textbf{receveur} qui déterminent
la compatibilité, pas ceux du donneur.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Retrouver les anticorps d'un groupe sanguin}
Identifier les antigènes présents, puis en déduire les anticorps absents (anti- les
antigènes non présents).
\begin{exemple}
Groupe B : antigène B présent $\Rightarrow$ anticorps anti-A présent (car pas d'antigène
A), pas d'anticorps anti-B.
\end{exemple}

\methode{Déterminer si une transfusion est possible}
Vérifier que le receveur ne possède pas d'anticorps contre les antigènes portés par les
globules rouges du donneur.
\begin{exemple}
Un donneur A vers un receveur B est impossible : le receveur B possède des anticorps
anti-A, qui agglutineraient les globules rouges transfusés.
\end{exemple}

\methode{Appliquer la règle du rhésus}
Vérifier la compatibilité ABO, puis vérifier que le receveur Rh$-$ ne reçoit que du sang
Rh$-$.
\begin{exemple}
Un donneur O Rh$+$ ne peut pas donner à un receveur Rh$-$, même si les groupes ABO sont
compatibles.
\end{exemple}
