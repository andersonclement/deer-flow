\chapter{La chronologie en géologie}

\noindent\textit{Comment savoir qu'une couche de roche est plus ancienne qu'une autre, ou
donner un âge précis à un fossile ? Les géologues disposent de deux grandes approches,
complémentaires : la datation relative, qui établit un ordre, et la datation absolue, qui
donne un âge en années.}
\vspace{8pt}

\connecteBanner

\section{La datation relative}

\begin{definition}[Datation relative]
La \textbf{datation relative} permet d'établir l'\textbf{ordre chronologique} d'événements
géologiques \textbf{sans donner d'âge précis} : elle indique qu'un événement est plus
ancien ou plus récent qu'un autre.
\end{definition}

\begin{propriete}[Principe de superposition]
Dans une série de \textbf{strates sédimentaires non perturbée}, les couches les plus
\textbf{anciennes} se trouvent \textbf{en bas}, et les plus \textbf{récentes} se trouvent
\textbf{en haut} : chaque nouvelle couche se dépose au-dessus des précédentes.
\end{propriete}

\begin{propriete}[Fossiles stratigraphiques]
Certains \textbf{fossiles}, caractéristiques d'une \textbf{courte période} géologique et
largement répandus géographiquement, servent de \textbf{fossiles stratigraphiques} (ou
guides) : leur présence dans deux couches de régions différentes permet d'affirmer
qu'elles se sont déposées à la même époque.
\end{propriete}

\begin{propriete}[Principe de recoupement]
Une structure géologique (faille, filon) qui en \textbf{recoupe} une autre est
nécessairement \textbf{plus récente} que celle qu'elle recoupe.
\end{propriete}

\section{La datation absolue}

\begin{definition}[Datation absolue]
La \textbf{datation absolue} donne un \textbf{âge précis, en années}, généralement grâce
à la \textbf{radioactivité} : certains éléments chimiques radioactifs se désintègrent à
une vitesse \textbf{constante et connue}.
\end{definition}

\begin{definition}[Demi-vie]
La \textbf{demi-vie} (ou période radioactive) $T$ d'un élément radioactif est la
\textbf{durée nécessaire} pour que la \textbf{moitié} des atomes radioactifs initialement
présents se soient désintégrés. Après $n$ demi-vies, il reste une fraction
$\left(\dfrac12\right)^n$ de la quantité initiale.
\end{definition}

\fig{Figure — Décroissance radioactive : la quantité restante est divisée par deux à chaque demi-vie.}
\end{illus}

\begin{exemple}
Le carbone 14, utilisé pour dater des matières organiques (bois, os) de quelques milliers
d'années, a une demi-vie $T=5\,730$ ans. Après $n=2$ demi-vies, soit
$t=2\times5\,730=11\,460$ ans, il ne reste plus qu'un quart de la quantité initiale de
carbone 14.
\end{exemple}

\begin{remarque}
Pour des roches très anciennes (millions ou milliards d'années), on utilise d'autres
couples d'éléments radioactifs, à demi-vie beaucoup plus longue (uranium-plomb,
potassium-argon), adaptés à ces échelles de temps considérables.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Datation relative.} Établit un ordre chronologique (superposition, fossiles
stratigraphiques, recoupement), sans âge précis.\\[4pt]
\textbf{Principe de superposition.} Bas $=$ ancien ; haut $=$ récent (strates non
perturbées).\\[4pt]
\textbf{Datation absolue.} Âge précis, en années, basé sur la radioactivité.\\[4pt]
\textbf{Demi-vie $T$.} Après $n$ demi-vies, il reste $\left(\dfrac12\right)^n$ de la
quantité initiale.\\[4pt]
\textbf{Carbone 14.} $T=5\,730$ ans, utilisé pour les matières organiques récentes.\\[4pt]
\textbf{Piège fréquent.} Après $2$ demi-vies, il ne reste \textbf{pas} $0\,\%$ mais
$\dfrac14$ ($25\,\%$) de la quantité initiale : la décroissance est
\textbf{exponentielle}, pas linéaire.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Ordonner des strates grâce au principe de superposition}
Dans une série non perturbée, la couche la plus basse est la plus ancienne.
\begin{exemple}
Si la strate C est sous la strate B, elle-même sous la strate A, l'ordre du plus ancien
au plus récent est C, B, A.
\end{exemple}

\methode{Calculer la fraction restante après $n$ demi-vies}
Appliquer $\left(\dfrac12\right)^n$.
\begin{exemple}
Après $n=3$ demi-vies : $\left(\dfrac12\right)^3=\dfrac18$ de la quantité initiale.
\end{exemple}

\methode{Calculer un âge à partir du nombre de demi-vies}
Appliquer $t=n\times T$, avec $T$ la demi-vie de l'élément utilisé.
\begin{exemple}
Carbone 14 ($T=5\,730$ ans), $n=3$ demi-vies : $t=3\times5\,730=17\,190$ ans.
\end{exemple}

\section{Sujets type BEPC}

\sujetbac[Principe de superposition et fossiles]
\begin{enumerate}[label=\arabic*)]
\item Énonce le principe de superposition.
\item Qu'est-ce qu'un fossile stratigraphique et à quoi sert-il ?
\item Trois strates non perturbées sont observées, de haut en bas : X, Y, Z. Laquelle
est la plus ancienne ?
\item Qu'apporte le principe de recoupement à la datation relative ?
\end{enumerate}

\sujetbac[Demi-vie et datation absolue]
\begin{enumerate}[label=\arabic*)]
\item Définis la demi-vie d'un élément radioactif.
\item Le carbone 14 a une demi-vie de $5\,730$ ans. Calcule l'âge correspondant à $2$
demi-vies.
\item Après ces $2$ demi-vies, quelle fraction de la quantité initiale de carbone 14
reste-t-il ?
\item Pourquoi la datation absolue permet-elle d'obtenir un âge précis, contrairement à
la datation relative ?
\end{enumerate}

\sujetbac[Comparer les deux méthodes de datation]
\begin{enumerate}[label=\arabic*)]
\item Quelle est la différence essentielle entre datation relative et datation
absolue ?
\item Cite une méthode de datation relative.
\item Cite le principe physique sur lequel repose la datation absolue.
\item Pourquoi ces deux méthodes sont-elles complémentaires plutôt que concurrentes ?
\end{enumerate}

\corrige{du sujet 3}
1) La datation relative établit un ordre chronologique ; la datation absolue donne un
âge précis en années.\\
2) Par exemple : le principe de superposition.\\
3) La désintégration radioactive (demi-vie).\\
4) Car la datation relative situe les événements les uns par rapport aux autres, tandis
que la datation absolue leur donne un âge précis : les deux approches se complètent pour
reconstituer l'histoire géologique.
