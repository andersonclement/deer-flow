\chapter{La vision et les organes des sens}

\noindent\textit{L'œil est le récepteur sensoriel de la vue : il transforme la lumière en
message nerveux. Comme tout instrument optique, il peut présenter des défauts, que des
lunettes permettent de corriger. Ce chapitre étudie son fonctionnement, puis passe en
revue les autres organes des sens.}
\vspace{8pt}

\connecteBanner

\section{Structure et fonctionnement de l'œil}

\begin{definition}[L'œil, récepteur de la vue]
L'\textbf{œil} est le récepteur sensoriel de la vue. La lumière le traverse dans l'ordre
suivant : \textbf{cornée} $\to$ \textbf{pupille} (ouverture centrale de l'\textbf{iris})
$\to$ \textbf{cristallin} (lentille convergente) $\to$ \textbf{rétine}, qui contient les
cellules photoréceptrices (cônes et bâtonnets). La rétine transforme la lumière reçue en
message nerveux, transmis au cerveau par le \textbf{nerf optique}.
\end{definition}

\begin{propriete}[Rôle de l'iris]
L'\textbf{iris} régule la quantité de lumière qui entre dans l'œil en modifiant le
diamètre de la \textbf{pupille} : celle-ci se rétrécit en pleine lumière et s'élargit
dans l'obscurité.
\end{propriete}

\section{L'accommodation}

\begin{definition}[Accommodation]
L'\textbf{accommodation} est la capacité du \textbf{cristallin} à modifier sa courbure
pour faire converger les rayons lumineux exactement sur la rétine, quelle que soit la
distance de l'objet observé. Pour un objet proche, le cristallin se bombe (devient plus
convergent) ; pour un objet éloigné, il s'aplatit.
\end{definition}

\begin{remarque}
L'accommodation est un phénomène musculaire (muscles ciliaires) : elle se fatigue, ce qui
explique la fatigue visuelle après une lecture prolongée.
\end{remarque}

\section{Les défauts de la vision}

\begin{propriete}[Trois défauts courants]
\begin{itemize}
\item \textbf{Myopie} : l'image se forme \textbf{avant} la rétine (œil trop long ou
cristallin trop convergent) ; la vision de loin est floue. Corrigée par une lentille
\textbf{divergente}.
\item \textbf{Hypermétropie} : l'image se forme \textbf{après} la rétine (œil trop court) ;
la vision de près est difficile. Corrigée par une lentille \textbf{convergente}.
\item \textbf{Presbytie} : le cristallin perd son élasticité avec l'âge et accommode mal
de près. Corrigée par une lentille \textbf{convergente} (verres de lecture).
\end{itemize}
\end{propriete}

\begin{exemple}
Une personne qui voit flou de loin mais nettement de près souffre probablement de
myopie : ses lunettes correctrices portent des verres divergents.
\end{exemple}

\section{Les autres organes des sens}

\begin{propriete}[Cinq sens, cinq récepteurs]
\begin{itemize}
\item \textbf{Vue} : récepteur $=$ rétine de l'œil (stimulus : lumière).
\item \textbf{Ouïe} : récepteur $=$ cellules ciliées de l'oreille interne (stimulus :
son).
\item \textbf{Toucher} : récepteur $=$ terminaisons nerveuses de la peau (stimulus :
pression, température, douleur).
\item \textbf{Odorat} : récepteur $=$ cellules olfactives du nez (stimulus : molécules
odorantes).
\item \textbf{Goût} : récepteur $=$ papilles gustatives de la langue (stimulus : molécules
sapides).
\end{itemize}
\end{propriete}

\begin{propriete}[Hygiène de la vue]
Préserver sa vue passe par : éviter une exposition prolongée aux écrans sans pause,
s'exposer à une lumière suffisante pour lire, et consulter régulièrement un
ophtalmologiste pour dépister un défaut de vision.
\end{propriete}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Trajet de la lumière.} Cornée $\to$ pupille (iris) $\to$ cristallin $\to$
rétine $\to$ nerf optique $\to$ cerveau.\\[4pt]
\textbf{Accommodation.} Le cristallin change de courbure pour faire la mise au point.\\[4pt]
\textbf{Myopie.} Image avant la rétine, corrigée par lentille divergente.
\textbf{Hypermétropie.} Image après la rétine, corrigée par lentille convergente.\\[4pt]
\textbf{Cinq sens.} Vue (rétine), ouïe (oreille interne), toucher (peau), odorat (nez),
goût (langue).\\[4pt]
\textbf{Piège fréquent.} Ne pas confondre myopie (image \textbf{avant} la rétine, corrigée
par lentille \textbf{divergente}) et hypermétropie (image \textbf{après}, lentille
\textbf{convergente}).
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Retracer le trajet de la lumière dans l'œil}
Suivre l'ordre : cornée, pupille, cristallin, rétine, nerf optique, cerveau.
\begin{exemple}
La lumière traverse d'abord la cornée, jamais directement la rétine.
\end{exemple}

\methode{Identifier un défaut de vision à partir d'un schéma}
Repérer la position de l'image formée par rapport à la rétine : avant $\to$ myopie ;
après $\to$ hypermétropie.
\begin{exemple}
Si l'image se forme avant la rétine sur un schéma, il s'agit d'une myopie.
\end{exemple}

\methode{Associer un sens à son récepteur}
Retrouver l'organe récepteur correspondant à chaque sens : vue (rétine), ouïe (oreille
interne), toucher (peau), odorat (nez), goût (langue).
\begin{exemple}
Le récepteur du goût est constitué par les papilles gustatives de la langue.
\end{exemple}
