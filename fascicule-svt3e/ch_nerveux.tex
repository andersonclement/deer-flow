\chapter{Le système nerveux et la fonction de relation}

\noindent\textit{Toucher un objet brûlant, entendre son nom, retirer la main d'un clou
pointu : à chaque instant, notre organisme perçoit des informations venues du milieu
extérieur et y répond. Cette capacité à percevoir et réagir s'appelle la fonction de
relation, assurée par le système nerveux.}
\vspace{8pt}

\connecteBanner

\section{Stimulus, récepteur et message nerveux}

\begin{definition}[Fonction de relation]
La \textbf{fonction de relation} est la capacité d'un organisme à percevoir des
informations de son environnement (ou de son propre corps) et à y répondre par un
comportement adapté.
\end{definition}

\begin{propriete}[Chaîne des éléments impliqués]
\begin{itemize}
\item Un \textbf{stimulus} (ou excitant) est un événement capable de déclencher une
réaction de l'organisme (lumière, son, piqûre, chaleur\dots).
\item Un \textbf{récepteur sensoriel} détecte le stimulus et le transforme en
\textbf{message nerveux} (œil, oreille, peau\dots).
\item Le message nerveux circule le long des \textbf{nerfs} jusqu'à un \textbf{centre
nerveux} (cerveau ou moelle épinière), qui l'analyse.
\item Le centre nerveux envoie un message nerveux moteur vers un \textbf{effecteur}
(muscle ou glande), qui exécute la réponse.
\end{itemize}
\end{propriete}

\section{L'acte réflexe}

\begin{definition}[Acte réflexe]
Un \textbf{acte réflexe} est une réaction \textbf{rapide, involontaire, inconsciente et
stéréotypée} (toujours identique pour un même stimulus). Exemple : retirer brusquement la
main d'un objet brûlant.
\end{definition}

\begin{propriete}[Les cinq éléments de l'arc réflexe]
Le trajet parcouru par le message nerveux lors d'un réflexe s'appelle l'\textbf{arc
réflexe}. Il comprend toujours cinq éléments : \textbf{récepteur} $\to$ \textbf{voie
sensitive (afférente)} $\to$ \textbf{centre nerveux} (souvent la moelle épinière) $\to$
\textbf{voie motrice (efférente)} $\to$ \textbf{effecteur}.
\end{propriete}

\begin{remarque}
Contrairement à un acte volontaire, le réflexe médullaire ne passe \textbf{pas} par le
cerveau : c'est ce qui explique sa très grande rapidité, utile pour protéger l'organisme
d'un danger immédiat.
\end{remarque}

\section{Réflexe et acte volontaire}

\begin{propriete}[Comparaison]
L'\textbf{acte volontaire} est \textbf{conscient}, contrôlé par le \textbf{cerveau}, et
peut varier d'une fois à l'autre. L'\textbf{acte réflexe} est \textbf{inconscient},
contrôlé le plus souvent par la \textbf{moelle épinière}, et toujours identique pour un
même stimulus.
\end{propriete}

\begin{exemple}
Attraper volontairement un ballon lancé est un acte volontaire (décision, visée, contrôle
du geste) ; retirer la main en touchant une casserole chaude est un acte réflexe
(automatique, avant même d'avoir « pensé » à la douleur).
\end{exemple}

\section{Vitesse de conduction du message nerveux}

\begin{propriete}[Vitesse du message nerveux]
Le message nerveux se propage le long d'un nerf à une \textbf{vitesse constante} $v$, qui
se calcule comme pour tout mouvement :
\[
v=\frac dt,
\]
avec $d$ la distance parcourue (en m) et $t$ la durée du trajet (en s), $v$ en m/s.
\end{propriete}

\begin{exemple}
Un message nerveux parcourt $d=0{,}9$ m (du pied à la moelle épinière) en $t=0{,}015$ s :
$v=\dfrac{0{,}9}{0{,}015}=60$ m/s.
\end{exemple}

\section{Hygiène du système nerveux}

\begin{propriete}[Préserver son système nerveux]
Le bon fonctionnement du système nerveux dépend de plusieurs habitudes de vie : un
\textbf{sommeil suffisant} (récupération des centres nerveux), une \textbf{alimentation
équilibrée}, et l'évitement de substances qui perturbent la transmission nerveuse
(\textbf{alcool}, \textbf{drogues}), qui ralentissent les réflexes et altèrent le
jugement.
\end{propriete}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Fonction de relation.} Percevoir un stimulus, y répondre par un comportement.\\[4pt]
\textbf{Trajet.} Récepteur $\to$ nerf sensitif $\to$ centre nerveux $\to$ nerf moteur
$\to$ effecteur.\\[4pt]
\textbf{Arc réflexe.} Récepteur, voie sensitive, centre nerveux, voie motrice, effecteur —
rapide, involontaire, inconscient, stéréotypé.\\[4pt]
\textbf{Acte volontaire.} Conscient, passe par le cerveau, variable.\\[4pt]
\textbf{Vitesse de conduction.} $v=\dfrac dt$.\\[4pt]
\textbf{Piège fréquent.} Un réflexe médullaire ne passe pas par le cerveau : ne pas
confondre « rapide » avec « traité par le cerveau ».
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Identifier les éléments d'un arc réflexe dans un schéma}
Repérer, dans l'ordre : récepteur, voie sensitive, centre nerveux, voie motrice,
effecteur.
\begin{exemple}
Dans le réflexe rotulien, le récepteur est un tendon du genou, le centre nerveux est la
moelle épinière, l'effecteur est le muscle de la cuisse.
\end{exemple}

\methode{Distinguer un acte réflexe d'un acte volontaire}
Se demander si la réaction est consciente et variable (volontaire) ou automatique et
toujours identique (réflexe).
\begin{exemple}
Cligner des yeux face à un objet qui approche brusquement est un réflexe ; choisir de
fermer les yeux pour se reposer est volontaire.
\end{exemple}

\methode{Calculer une vitesse de conduction nerveuse}
Appliquer $v=\dfrac dt$, avec $d$ en m et $t$ en s.
\begin{exemple}
$d=1{,}2$ m, $t=0{,}02$ s : $v=\dfrac{1{,}2}{0{,}02}=60$ m/s.
\end{exemple}
