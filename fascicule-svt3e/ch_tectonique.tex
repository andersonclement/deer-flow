\chapter{La tectonique des plaques}

\noindent\textit{Les continents ne sont pas immobiles : ils dérivent lentement, portés
par d'immenses plaques rocheuses qui recouvrent la surface du globe. Ce chapitre étudie
la structure interne de la Terre, ces plaques lithosphériques, et les phénomènes
géologiques qui se produisent à leurs limites.}
\vspace{8pt}

\connecteBanner

\section{La structure interne du globe}

\begin{propriete}[Trois grandes enveloppes]
De l'extérieur vers le centre, la Terre est constituée de la \textbf{croûte} (fine
couche superficielle, continentale ou océanique), du \textbf{manteau} (épaisse couche
rocheuse, en partie ductile), et du \textbf{noyau} (noyau externe liquide, noyau interne
solide, riches en fer).
\end{propriete}

\section{Les plaques lithosphériques}

\begin{definition}[Lithosphère et plaques]
La \textbf{lithosphère} est la couche rigide superficielle de la Terre (croûte $+$
sommet du manteau), découpée en une dizaine de grandes \textbf{plaques
lithosphériques}. Ces plaques « flottent » et se déplacent lentement sur
l'\textbf{asthénosphère}, une couche du manteau plus ductile située en dessous.
\end{definition}

\begin{propriete}[Une dérive très lente]
Les plaques se déplacent de quelques \textbf{centimètres par an}, un mouvement
imperceptible à l'échelle humaine mais considérable à l'échelle des temps géologiques.
Cette théorie confirme et explique la \textbf{dérive des continents}, proposée dès 1912
par Alfred Wegener.
\end{propriete}

\begin{exemple}
Une plaque se déplaçant à $v=2$ cm/an parcourt, en $t=1\,000\,000$ d'années, une
distance $d=v\times t=2\,000\,000$ cm $=20$ km.
\end{exemple}

\section{Les limites de plaques}

\begin{propriete}[Trois types de limites]
\begin{itemize}
\item \textbf{Limite divergente} (dorsale océanique) : les plaques s'\textbf{écartent} ;
du magma remonte et forme de la \textbf{nouvelle croûte océanique}.
\item \textbf{Limite convergente} : les plaques se \textbf{rapprochent}. Si une plaque
océanique rencontre une plaque continentale (ou une autre plaque océanique), elle plonge
sous l'autre : c'est la \textbf{subduction} (fosse océanique, volcans, séismes). Si deux
plaques continentales se rencontrent, elles se \textbf{soulèvent} : c'est une
\textbf{collision} (formation d'une chaîne de montagnes, comme l'Himalaya).
\item \textbf{Limite coulissante} (faille transformante) : les plaques
\textbf{glissent} latéralement l'une contre l'autre, sans création ni destruction de
croûte, mais avec de fréquents \textbf{séismes}.
\end{itemize}
\end{propriete}

\begin{remarque}
Séismes et volcans ne sont pas répartis au hasard sur le globe : ils se concentrent très
majoritairement aux \textbf{limites de plaques}, dessinant des zones à risque bien
identifiées (comme la « ceinture de feu » qui entoure l'océan Pacifique).
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Structure interne.} Croûte, manteau, noyau externe (liquide), noyau interne
(solide).\\[4pt]
\textbf{Lithosphère.} Couche rigide découpée en plaques, qui se déplacent sur
l'asthénosphère.\\[4pt]
\textbf{Vitesse.} Quelques cm/an ; $d=v\times t$.\\[4pt]
\textbf{Trois limites.} Divergente (écartement, dorsale) ; convergente (subduction ou
collision) ; coulissante (glissement, séismes).\\[4pt]
\textbf{Répartition.} Séismes et volcans se concentrent aux limites de plaques.\\[4pt]
\textbf{Piège fréquent.} Une limite convergente ne produit pas toujours des volcans : la
collision continentale (Himalaya) forme des montagnes sans volcanisme notable, contrairement
à la subduction.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer une distance de déplacement d'une plaque}
Appliquer $d=v\times t$, en veillant à la cohérence des unités.
\begin{exemple}
$v=5$ cm/an, $t=2\,000\,000$ ans : $d=10\,000\,000$ cm $=100$ km.
\end{exemple}

\methode{Identifier un type de limite de plaques}
Observer si les plaques s'écartent (divergente), se rapprochent (convergente) ou
glissent latéralement (coulissante).
\begin{exemple}
Une dorsale océanique, où de la croûte se forme en continu, est une limite divergente.
\end{exemple}

\methode{Relier un phénomène géologique à un type de limite}
Associer volcans et fosses océaniques à la subduction ; montagnes sans volcans à la
collision ; séismes fréquents sans volcanisme à une limite coulissante.
\begin{exemple}
Une chaîne de montagnes sans activité volcanique évoque une collision continentale.
\end{exemple}
