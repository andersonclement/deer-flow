\chapter{Les chromosomes, support de l'information génétique}

\noindent\textit{Chaque cellule de notre corps contient, dans son noyau, l'ensemble de
l'information nécessaire à notre développement. Cette information est portée par des
structures appelées chromosomes. Leur nombre et leur organisation, observables au
microscope, permettent de détecter certaines anomalies dès avant la naissance.}
\vspace{8pt}

\connecteBanner

\section{Les chromosomes, structures du noyau cellulaire}

\begin{definition}[Chromosome]
Un \textbf{chromosome} est une structure filamenteuse, visible au microscope pendant la
division cellulaire, présente dans le \textbf{noyau} de chaque cellule. Il est constitué
d'\textbf{ADN} (acide désoxyribonucléique), la molécule qui porte l'information
génétique.
\end{definition}

\begin{propriete}[Chromosome dupliqué]
Avant une division cellulaire, chaque chromosome se duplique : il est alors formé de
deux \textbf{chromatides} identiques, reliées en un point appelé
\textbf{centromère}.
\end{propriete}

\section{Le caryotype humain}

\begin{definition}[Caryotype]
Le \textbf{caryotype} est la représentation ordonnée de l'ensemble des chromosomes
d'une cellule, classés par paires de taille décroissante. Il est établi à partir d'une
cellule en division (prélèvement sanguin ou, avant la naissance, prélèvement de
liquide amniotique).
\end{definition}

\begin{propriete}[Caryotype humain normal]
Une cellule humaine normale contient \textbf{46 chromosomes}, soit \textbf{23 paires} :
\begin{itemize}
\item \textbf{22 paires d'autosomes}, identiques chez l'homme et chez la femme ;
\item \textbf{1 paire de chromosomes sexuels} : \textbf{XX} chez la femme, \textbf{XY}
chez l'homme.
\end{itemize}
\end{propriete}

\begin{exemple}
Un caryotype montrant 44 autosomes + XY (soit 46 chromosomes au total) est celui d'un
homme. Un caryotype montrant 44 autosomes + XX est celui d'une femme.
\end{exemple}

\section{Les anomalies chromosomiques}

\begin{definition}[Anomalie chromosomique]
Une \textbf{anomalie chromosomique} est une modification du nombre ou de la structure
des chromosomes, visible sur le caryotype. Elle survient accidentellement lors de la
formation des cellules reproductrices ou lors de la fécondation.
\end{definition}

\begin{propriete}[La trisomie 21]
La \textbf{trisomie 21} est l'anomalie chromosomique numérique la plus fréquente : la
paire de chromosomes n°21 est remplacée par \textbf{trois} chromosomes 21 au lieu de
deux. Le caryotype compte alors \textbf{47 chromosomes} au lieu de 46. Elle entraîne un
retard du développement intellectuel et des traits physiques caractéristiques.
\end{propriete}

\fig{Figure — Trisomie 21 : trois exemplaires du chromosome 21 au lieu de deux (47

\end{illus}

\begin{exemple}
Le diagnostic prénatal (échographie, dosages sanguins, et si besoin analyse du
caryotype à partir du liquide amniotique) permet de détecter une trisomie 21 avant la
naissance.
\end{exemple}

\begin{remarque}
Piège fréquent : la trisomie 21 touche une seule paire de chromosomes (le n°21) ; les
44 autres chromosomes restent en nombre normal. Le caryotype complet compte donc 47
chromosomes, et non 48 ou 45.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Chromosome.} Structure du noyau, constituée d'ADN, visible pendant la division
cellulaire.\\[4pt]
\textbf{Chromosome dupliqué.} Deux chromatides identiques reliées par le
centromère.\\[4pt]
\textbf{Caryotype.} Classement ordonné des chromosomes d'une cellule.\\[4pt]
\textbf{Caryotype humain normal.} 46 chromosomes = 23 paires (22 paires d'autosomes +
1 paire de chromosomes sexuels XX ou XY).\\[4pt]
\textbf{Trisomie 21.} Anomalie chromosomique numérique : 3 chromosomes 21 au lieu de 2,
soit 47 chromosomes au total.\\[4pt]
\textbf{Piège fréquent.} Seule la paire concernée est en surnombre ; les autres
chromosomes restent normaux.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Lire un caryotype}
Compter le nombre total de chromosomes, identifier les paires d'autosomes (numérotées)
et la paire de chromosomes sexuels (XX ou XY) pour déterminer le sexe et détecter une
éventuelle anomalie.
\begin{exemple}
Un caryotype avec 44 autosomes + XY compte 46 chromosomes : c'est un caryotype
masculin normal.
\end{exemple}

\methode{Identifier une trisomie sur un caryotype}
Chercher une paire représentée par \textbf{trois} chromosomes au lieu de deux ; le
numéro de cette paire donne le nom de la trisomie.
\begin{exemple}
Trois chromosomes 21 au lieu de deux : il s'agit d'une trisomie 21, caryotype à 47
chromosomes.
\end{exemple}

\section{Corrigés}

\corrige{de l'exercice 1}
D'ADN, organisé en une structure filamenteuse visible pendant la division cellulaire.

\corrige{de l'exercice 2}
Le point qui relie les deux chromatides d'un chromosome dupliqué.

\corrige{de l'exercice 3}
46 chromosomes.

\corrige{de l'exercice 4}
La paire de chromosomes sexuels (XX ou XY).

\corrige{de l'exercice 5}
C'est le caryotype d'une femme.

\corrige{de l'exercice 6}
47 chromosomes.

\corrige{de l'exercice 7}
La paire n°21.

\corrige{de l'exercice 8}
Car ce n'est que pendant la division cellulaire que les chromosomes, dupliqués et
condensés, deviennent suffisamment visibles au microscope pour être comptés et
classés ; au repos, l'ADN est trop diffus dans le noyau.

\corrige{de l'exercice 9}
Cette information porte sur la 23\textsuperscript{e} paire de chromosomes (les
chromosomes sexuels) : XX pour une fille, XY pour un garçon.

\corrige{de l'exercice 10}
Non : dans une trisomie 21, c'est précisément la paire 21 qui compte trois
exemplaires ; ici, c'est une autre paire d'autosomes qui est en surnombre, il ne s'agit
donc pas d'une trisomie 21.

\corrige{de l'exercice 11}
Parce qu'il permet aux parents et à l'équipe médicale de se préparer et d'organiser,
dès la naissance, un suivi médical et un accompagnement adaptés à l'enfant.

\corrige{de l'exercice 12}
C'est faux : la trisomie 21 correspond au contraire à un chromosome \textbf{en trop}
(trois chromosomes 21 au lieu de deux), et non à un chromosome manquant.
