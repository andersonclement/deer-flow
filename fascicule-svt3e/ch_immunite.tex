\chapter{La réponse immunitaire}

\noindent\textit{Notre organisme est en permanence exposé à des microbes
potentiellement dangereux. Il dispose pourtant de moyens de défense efficaces, qui
s'organisent en deux grandes réponses complémentaires : une réponse rapide et
générale, la réponse non spécifique, et une réponse plus lente mais ciblée, la réponse
spécifique.}
\vspace{8pt}

\connecteBanner

\section{Les barrières naturelles, première protection}

\begin{definition}[Immunité]
L'\textbf{immunité} est l'ensemble des mécanismes par lesquels l'organisme se
\textbf{défend} contre les agents pathogènes (bactéries, virus, champignons\dots) et
les corps étrangers.
\end{definition}

\begin{propriete}[Peau et muqueuses]
La \textbf{peau} et les \textbf{muqueuses} (yeux, bouche, voies respiratoires) forment
une barrière physique et chimique qui empêche la plupart des microbes de pénétrer dans
l'organisme. Lorsque cette barrière est franchie (plaie, piqûre), d'autres défenses
entrent en jeu.
\end{propriete}

\section{La réponse non spécifique : les phagocytes}

\begin{definition}[Réponse non spécifique]
La \textbf{réponse non spécifique} est une défense \textbf{immédiate}, présente dès la
naissance, qui réagit de la \textbf{même façon} face à n'importe quel agent pathogène,
sans le distinguer précisément.
\end{definition}

\begin{definition}[Réaction inflammatoire]
La \textbf{réaction inflammatoire} est une réponse locale et rapide de l'organisme
face à une infection ou une blessure. Elle se manifeste par quatre signes
caractéristiques : \textbf{rougeur}, \textbf{chaleur}, \textbf{douleur} et
\textbf{gonflement}.
\end{definition}

\begin{propriete}[Mécanisme]
Au site de l'infection, les vaisseaux sanguins se dilatent (d'où la rougeur et la
chaleur) et deviennent plus perméables, ce qui permet à de nombreux \textbf{phagocytes}
(globules blancs) de sortir du sang et de converger vers la zone infectée.
\end{propriete}

\begin{definition}[Phagocytose]
La \textbf{phagocytose} est le mécanisme par lequel un phagocyte \textbf{englobe} puis
\textbf{détruit} un microbe.
\end{definition}

\begin{remarque}
La fièvre, souvent associée à une infection, aide également l'organisme à se défendre :
une température plus élevée ralentit la multiplication de nombreux microbes.
\end{remarque}

\section{La réponse spécifique : les lymphocytes}

\begin{definition}[Réponse spécifique]
La \textbf{réponse spécifique} fait intervenir des cellules appelées
\textbf{lymphocytes}, capables de reconnaître un agent pathogène \textbf{précis} et de
le combattre de façon ciblée. Elle se met en place plus \textbf{lentement} que la
réponse non spécifique (quelques jours), mais elle est beaucoup plus efficace contre
l'agent identifié.
\end{definition}

\begin{propriete}[Le rôle des lymphocytes]
Face à un microbe précis, certains lymphocytes se multiplient et produisent des
\textbf{anticorps}, des molécules qui se fixent spécifiquement sur ce microbe et
facilitent sa destruction.
\end{propriete}

\begin{propriete}[La mémoire immunitaire]
Après une première rencontre avec un microbe, certains lymphocytes persistent dans
l'organisme sous forme de \textbf{cellules mémoires}. Lors d'un second contact avec le
\textbf{même} microbe, la réponse spécifique est alors \textbf{beaucoup plus rapide et
plus efficace}.
\end{propriete}

\begin{exemple}
C'est grâce à la mémoire immunitaire qu'une personne ayant déjà eu la varicelle ne la
contracte, en général, plus une seconde fois : ses lymphocytes mémoires reconnaissent
immédiatement le virus lors d'un nouveau contact.
\end{exemple}

\begin{remarque}
Piège fréquent : la réponse non spécifique agit contre \textbf{tous} les
microorganismes de la même façon, alors que la réponse spécifique cible un microbe
\textbf{précis} et laisse, en plus, une trace durable (la mémoire immunitaire) que la
réponse non spécifique ne laisse pas.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Immunité.} Ensemble des défenses de l'organisme contre les agents
pathogènes.\\[4pt]
\textbf{Barrières naturelles.} Peau et muqueuses, première protection.\\[4pt]
\textbf{Réponse non spécifique.} Immédiate, identique pour tout agent pathogène :
réaction inflammatoire (rougeur, chaleur, douleur, gonflement) et phagocytose.\\[4pt]
\textbf{Réponse spécifique.} Plus lente, ciblée sur un agent précis : lymphocytes,
anticorps.\\[4pt]
\textbf{Mémoire immunitaire.} Cellules mémoires permettant une réponse spécifique plus
rapide lors d'un second contact avec le même microbe.\\[4pt]
\textbf{Piège fréquent.} Seule la réponse spécifique est ciblée et laisse une mémoire ;
la réponse non spécifique agit de la même façon contre tout agent pathogène.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Reconnaître les signes d'une réaction inflammatoire}
Rechercher les quatre signes caractéristiques : rougeur, chaleur, douleur, gonflement.
\begin{exemple}
Une zone rouge, chaude et gonflée autour d'une écharde correspond à une réaction
inflammatoire.
\end{exemple}

\methode{Distinguer réponse non spécifique et réponse spécifique}
Se demander si la réponse est immédiate et identique pour tout agent pathogène (non
spécifique) ou ciblée sur un microbe précis, avec mémoire (spécifique).
\begin{exemple}
La phagocytose, qui agit contre tout microbe, relève de la réponse non spécifique ; la
production d'anticorps contre un virus précis relève de la réponse spécifique.
\end{exemple}

\corrige{de l'exercice 7}

\corrige{de l'exercice 10}
Parce qu'elle permet de mobiliser rapidement les phagocytes vers la zone infectée pour

\corrige{de l'exercice 11}
Car une température plus élevée ralentit la multiplication de nombreux microbes.

\corrige{de l'exercice 12}
Une réaction inflammatoire, c'est-à-dire un afflux de phagocytes destiné à combattre
une infection locale.

\corrige{de l'exercice 13}
Car elle réagit de la même façon, immédiatement, face à n'importe quel agent
pathogène, sans le distinguer précisément.

\corrige{de l'exercice 14}
Car elle nécessite la reconnaissance précise de l'agent pathogène puis la
multiplication des lymphocytes capables de le combattre, ce qui prend plusieurs
jours.

\corrige{de l'exercice 15}
Parce que des cellules mémoires, issues du premier contact, sont déjà présentes dans
l'organisme et permettent une réponse spécifique immédiate et efficace dès le second
contact.

\corrige{du sujet 1}
1) Rougeur, chaleur, douleur, gonflement.\\
2) Le mécanisme par lequel un phagocyte englobe puis détruit un microbe.\\
3) Pour permettre un afflux plus important de sang et de phagocytes vers la zone
infectée.\\
4) Elle permet de mobiliser rapidement les défenses de l'organisme contre
l'infection.

\corrige{du sujet 2}
1) Une réponse immédiate et identique face à tout agent pathogène (phagocytose,
réaction inflammatoire).\\
2) Une réponse plus lente, ciblée sur un agent pathogène précis, faisant intervenir
des lymphocytes et des anticorps.\\
3) Par exemple : la rapidité (réponse non spécifique) contre la spécificité (réponse
spécifique).\\
4) Elle permet, grâce aux cellules mémoires, une réponse spécifique plus rapide et
plus efficace lors d'un second contact avec le même microbe.
