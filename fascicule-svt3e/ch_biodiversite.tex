\chapter{Biodiversité, environnement, hygiène et biotechnologie}

\noindent\textit{Ce dernier chapitre relie plusieurs fils du fascicule : la diversité du
vivant qui nous entoure, les menaces qui pèsent sur elle, les bons réflexes d'hygiène de
vie, et la manière dont l'être humain met les êtres vivants au service de ses besoins,
grâce à la biotechnologie.}
\vspace{8pt}

\connecteBanner

\section{La biodiversité}

\begin{definition}[Biodiversité]
La \textbf{biodiversité} désigne la diversité du vivant, à trois niveaux : la
\textbf{diversité des écosystèmes} (forêts, savanes, océans\dots), la \textbf{diversité
des espèces} au sein de ces écosystèmes, et la \textbf{diversité génétique} au sein
d'une même espèce (voir le chapitre sur l'hérédité).
\end{definition}

\begin{propriete}[Des menaces majoritairement humaines]
La biodiversité est aujourd'hui menacée par plusieurs facteurs, essentiellement
d'origine humaine : la \textbf{déforestation}, la \textbf{pollution} (eau, air, sols),
la \textbf{surexploitation} (surpêche, braconnage), le \textbf{changement climatique}, et
l'introduction d'\textbf{espèces invasives} qui déséquilibrent les écosystèmes locaux.
\end{propriete}

\begin{exemple}
Une population animale passe de $V_i=2\,000$ individus à $V_f=500$ individus en
quelques années. Le taux de variation est
$t=\dfrac{V_f-V_i}{V_i}\times100=\dfrac{500-2\,000}{2\,000}\times100=-75\,\%$ : la
population a chuté de $75\,\%$.
\end{exemple}

\begin{propriete}[Protéger la biodiversité]
Plusieurs actions permettent de limiter l'érosion de la biodiversité : la création
d'\textbf{aires protégées} (parcs nationaux, réserves naturelles), des \textbf{lois} de
protection des espèces menacées, le \textbf{reboisement}, la réduction de la
\textbf{pollution}, et le développement d'une \textbf{agriculture durable}.
\end{propriete}

\section{L'hygiène de vie}

\begin{propriete}[Des habitudes qui protègent la santé]
Une bonne hygiène de vie s'appuie sur plusieurs piliers étudiés tout au long de ce
fascicule : une \textbf{alimentation équilibrée}, une \textbf{activité physique}
régulière, un \textbf{sommeil suffisant} (essentiel au système nerveux, voir chapitre 1),
une \textbf{hygiène corporelle} adaptée, et le respect des mesures de \textbf{prévention}
contre les infections (vaccination, dépistage, voir chapitres 7 et 9).
\end{propriete}

\begin{remarque}
Ces habitudes agissent en synergie : un sommeil insuffisant affaiblit par exemple le
système immunitaire, rendant l'organisme plus vulnérable aux infections.
\end{remarque}

\section{La biotechnologie}

\begin{definition}[Biotechnologie]
La \textbf{biotechnologie} regroupe l'ensemble des techniques qui utilisent des êtres
vivants, ou des substances qu'ils produisent, pour créer des produits utiles à l'être
humain.
\end{definition}

\begin{propriete}[Des exemples déjà rencontrés]
\begin{itemize}
\item La \textbf{fermentation} par les levures (fabrication du pain, du yaourt, de
boissons fermentées — voir chapitres 4 et 11) ;
\item La production d'\textbf{antibiotiques} (comme la pénicilline), obtenus à partir
de certaines moisissures ;
\item La production d'\textbf{insuline} par des bactéries génétiquement modifiées, pour
traiter le diabète ;
\item Le \textbf{traitement des eaux usées} par des bactéries qui dégradent les matières
organiques polluantes.
\end{itemize}
\end{propriete}

\begin{exemple}
Avant les biotechnologies modernes, l'insuline utilisée pour traiter le diabète était
extraite d'organes animaux ; aujourd'hui, des bactéries modifiées la produisent en
laboratoire, en quantité bien plus importante et à moindre coût.
\end{exemple}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Biodiversité.} Trois niveaux : écosystèmes, espèces, diversité génétique.\\[4pt]
\textbf{Menaces.} Déforestation, pollution, surexploitation, changement climatique,
espèces invasives.\\[4pt]
\textbf{Protection.} Aires protégées, lois, reboisement, réduction de la pollution,
agriculture durable.\\[4pt]
\textbf{Hygiène de vie.} Alimentation, activité physique, sommeil, hygiène corporelle,
prévention.\\[4pt]
\textbf{Biotechnologie.} Utilisation des êtres vivants pour produire des biens utiles
(fermentation, antibiotiques, insuline, traitement des eaux).\\[4pt]
\textbf{Piège fréquent.} La biotechnologie n'est pas seulement une technologie
« moderne » : la fermentation, utilisée depuis des millénaires pour le pain ou le
fromage, en est déjà une forme.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Calculer un taux de variation d'une population}
Appliquer $t=\dfrac{V_f-V_i}{V_i}\times100$.
\begin{exemple}
$V_i=5\,000$, $V_f=3\,000$ : $t=\dfrac{3\,000-5\,000}{5\,000}\times100=-40\,\%$.
\end{exemple}

\methode{Relier une action humaine à une menace sur la biodiversité}
Identifier le facteur en jeu : destruction d'habitat (déforestation), rejets nocifs
(pollution), prélèvement excessif (surexploitation).
\begin{exemple}
La coupe massive d'une forêt tropicale relève de la déforestation.
\end{exemple}

\methode{Reconnaître un exemple de biotechnologie}
Vérifier qu'un être vivant (ou une substance qu'il produit) est utilisé pour fabriquer un
produit utile à l'être humain.
\begin{exemple}
La fabrication du yaourt à partir de bactéries lactiques est un exemple de
biotechnologie.
\end{exemple}

2) Par exemple : déforestation, pollution, surexploitation.\\
3) $t=\dfrac{500-2\,000}{2\,000}\times100=-75\,\%$.\\
4) Par exemple : créer une aire protégée et légiférer contre le braconnage.

\corrige{du sujet 2}
1) Alimentation équilibrée, activité physique, sommeil suffisant, hygiène corporelle.\\
2) Un sommeil suffisant permet une bonne récupération de l'organisme, favorisant
indirectement de bonnes défenses immunitaires.\\
3) Car elle prépare le système immunitaire à réagir rapidement en cas d'infection
réelle, réduisant le risque de maladie.\\
4) Par exemple : limiter le temps d'écran avant le coucher pour favoriser un bon
sommeil.

\corrige{du sujet 3}
1) L'ensemble des techniques qui utilisent des êtres vivants pour produire des biens
utiles à l'être humain.\\
2) Par exemple : la fermentation, la production d'antibiotiques, la production
d'insuline.\\
3) Elle est produite par des bactéries génétiquement modifiées cultivées en
laboratoire.\\
4) Car elle utilise des micro-organismes (les levures) pour transformer les aliments,
ce qui est le principe même de la biotechnologie.
