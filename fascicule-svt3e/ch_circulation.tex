\chapter{La circulation sanguine et les accidents cardiovasculaires}

\noindent\textit{Le cœur et les vaisseaux sanguins forment un circuit qui apporte en
permanence oxygène et nutriments à toutes les cellules du corps. Ce chapitre étudie
leur fonctionnement, les règles d'hygiène qui les préservent, et les principaux
accidents qui peuvent les affecter.}
\vspace{8pt}

\connecteBanner

\section{L'appareil circulatoire}

\begin{definition}[Appareil circulatoire]
L'\textbf{appareil circulatoire} est formé du \textbf{cœur}, une pompe musculaire, et
des \textbf{vaisseaux sanguins}, qui forment un circuit fermé permettant au sang de
circuler dans tout l'organisme.
\end{definition}

\begin{propriete}[Trois types de vaisseaux]
\begin{itemize}
\item les \textbf{artères} transportent le sang \textbf{du cœur vers les organes} ;
\item les \textbf{veines} ramènent le sang \textbf{des organes vers le cœur} ;
\item les \textbf{capillaires}, très fins, permettent les \textbf{échanges} entre le
sang et les cellules (apport de dioxygène et de nutriments, récupération de
déchets).
\end{itemize}
\end{propriete}

\begin{propriete}[Le fonctionnement du cœur]
Le cœur se contracte régulièrement (les \textbf{battements cardiaques}) pour propulser
le sang dans les artères. Ce rythme s'accélère à l'effort, pour répondre aux besoins
accrus des muscles en dioxygène et en nutriments.
\end{propriete}

\section{L'hygiène de l'appareil circulatoire}

\begin{propriete}[Préserver son cœur et ses vaisseaux]
\begin{itemize}
\item une \textbf{activité physique régulière}, qui renforce le muscle cardiaque ;
\item une \textbf{alimentation équilibrée}, limitée en graisses et en sel ;
\item éviter le \textbf{tabac}, qui abîme les parois des vaisseaux ;
\item limiter la \textbf{consommation d'alcool} ;
\item surveiller régulièrement sa \textbf{tension artérielle}.
\end{itemize}
\end{propriete}

\section{Les accidents cardiovasculaires}

\begin{definition}[Hémorragie]
Une \textbf{hémorragie} est un écoulement de sang hors des vaisseaux, dû à leur
rupture. Elle peut être \textbf{externe} (visible) ou \textbf{interne} (non
visible, mais parfois grave).
\end{definition}

\begin{definition}[Hypertension artérielle]
L'\textbf{hypertension artérielle} est une pression du sang \textbf{anormalement
élevée} dans les artères. Souvent silencieuse, elle fatigue le cœur et les vaisseaux
sur le long terme et constitue un facteur de risque majeur pour les autres accidents
cardiovasculaires.
\end{definition}

\begin{definition}[Artériosclérose]
L'\textbf{artériosclérose} est un \textbf{épaississement et un durcissement} de la
paroi des artères, souvent lié au dépôt de graisses (athérome), qui réduit le diamètre
utile du vaisseau et gêne la circulation du sang.
\end{definition}

\begin{definition}[Varices]
Les \textbf{varices} sont des veines, le plus souvent des jambes, anormalement
\textbf{dilatées} et déformées, dues à un mauvais retour du sang vers le cœur.
\end{definition}

\begin{definition}[Infarctus du myocarde]
L'\textbf{infarctus du myocarde} (« crise cardiaque ») survient lorsqu'une artère qui
irrigue le \textbf{muscle cardiaque} (le myocarde) est \textbf{obstruée} : la partie du
cœur privée de sang ne reçoit plus de dioxygène et commence à mourir.
\end{definition}

\begin{definition}[Accident vasculaire cérébral (AVC)]
Un \textbf{accident vasculaire cérébral} survient lorsqu'un vaisseau du
\textbf{cerveau} est \textbf{obstrué} ou se \textbf{rompt} : une partie du cerveau,
privée de sang, cesse de fonctionner et peut être définitivement endommagée.
\end{definition}

\begin{propriete}[Un point commun : l'obstruction]
L'infarctus du myocarde et de nombreux AVC ont une cause commune : l'obstruction
d'une artère, souvent favorisée par l'artériosclérose. Seul l'organe touché diffère
(le cœur ou le cerveau).
\end{propriete}

\begin{remarque}
Piège fréquent : infarctus et AVC ne touchent pas le même organe. L'infarctus affecte
le \textbf{cœur} lui-même (le muscle cardiaque) ; l'AVC affecte le \textbf{cerveau}.
Les deux peuvent néanmoins avoir la même cause sous-jacente, l'obstruction d'une
artère.
\end{remarque}

\begin{exemple}
Une personne présentant une hypertension artérielle non traitée et un tabagisme
important cumule deux facteurs de risque qui favorisent l'artériosclérose, et donc
l'infarctus et l'AVC.
\end{exemple}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Appareil circulatoire.} Cœur (pompe) + vaisseaux (artères, veines,
capillaires).\\[4pt]
\textbf{Artères.} Cœur $\to$ organes. \textbf{Veines.} Organes $\to$ cœur.\\[4pt]
\textbf{Hygiène.} Activité physique, alimentation équilibrée, pas de tabac,
surveillance de la tension.\\[4pt]
\textbf{Hypertension.} Pression sanguine anormalement élevée.\\[4pt]
\textbf{Artériosclérose.} Épaississement/durcissement des artères (dépôt de
graisses).\\[4pt]
\textbf{Infarctus du myocarde.} Obstruction d'une artère du cœur.\\[4pt]
\textbf{AVC.} Obstruction ou rupture d'un vaisseau du cerveau.\\[4pt]
\textbf{Piège fréquent.} Infarctus = cœur ; AVC = cerveau ; cause souvent commune :
l'obstruction d'une artère.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Distinguer infarctus et AVC}
Identifier l'organe touché par l'obstruction ou la rupture d'un vaisseau : le cœur
(infarctus) ou le cerveau (AVC).
\begin{exemple}
Une douleur thoracique brutale liée à l'obstruction d'une artère coronaire évoque un
infarctus ; une perte soudaine de la parole liée à l'atteinte d'un vaisseau cérébral
évoque un AVC.
\end{exemple}

\methode{Relier un facteur de risque à un accident cardiovasculaire}
Relier tabac, alimentation grasse/salée, sédentarité et hypertension à leur
conséquence commune : le développement de l'artériosclérose.
\begin{exemple}
Le tabac favorise l'artériosclérose, qui elle-même augmente le risque d'infarctus et
d'AVC.
\end{exemple}

\section{Corrigés}
\begin{multicols}{2}

\corrige{de l'exercice 1}
Les artères, les veines, les capillaires.

\corrige{de l'exercice 2}
L'artère transporte le sang du cœur vers les organes ; la veine ramène le sang des
organes vers le cœur.

\corrige{de l'exercice 3}
Par exemple : une activité physique régulière et une alimentation équilibrée.

\corrige{de l'exercice 4}
Une pression du sang anormalement élevée dans les artères.

\corrige{de l'exercice 5}
Un épaississement et un durcissement de la paroi des artères, souvent lié au dépôt de
graisses.

\corrige{de l'exercice 6}
L'infarctus touche le cœur ; l'AVC touche le cerveau.

\corrige{de l'exercice 7}
Un écoulement de sang hors des vaisseaux, dû à leur rupture.

\corrige{de l'exercice 8}
Car les muscles ont besoin de davantage de dioxygène et de nutriments pendant
l'effort ; le cœur bat donc plus vite pour augmenter le débit sanguin.

\corrige{de l'exercice 9}
Les deux résultent souvent de l'obstruction d'une artère, généralement favorisée par
l'artériosclérose ; seul l'organe touché diffère (cœur ou cerveau).

\corrige{de l'exercice 10}
Car il abîme les parois des vaisseaux et favorise le développement de
l'artériosclérose, qui peut ensuite obstruer une artère du cœur.

\corrige{de l'exercice 11}
Car elle fatigue progressivement le cœur et les vaisseaux, sans provoquer de
symptôme visible, avant de provoquer un accident cardiovasculaire grave.

\corrige{de l'exercice 12}
C'est faux : un infarctus touche le cœur (le muscle cardiaque) ; c'est l'AVC qui touche
le cerveau.

\corrige{de l'exercice 13}
Car la position debout prolongée et la gravité gênent le retour du sang vers le
cœur dans les veines des jambes, favorisant leur dilatation.

\corrige{du sujet 1}
1) Artères (cœur $\to$ organes), veines (organes $\to$ cœur), capillaires
(échanges).\\
2) Propulser le sang dans les vaisseaux par ses contractions rythmiques.\\
3) Par exemple : activité physique régulière et alimentation équilibrée.

\corrige{du sujet 2}
1) Hypertension : pression sanguine anormalement élevée. Artériosclérose :
épaississement/durcissement des artères, souvent par dépôt de graisses.\\
2) Une artère qui irrigue le cœur s'obstrue : la partie du muscle cardiaque privée de
sang commence à mourir.\\
3) Un vaisseau du cerveau s'obstrue ou se rompt : une partie du cerveau, privée de
sang, cesse de fonctionner.\\
4) Par exemple : le tabac et l'hypertension artérielle.
