\chapter{Les microorganismes de notre environnement et la contamination}

\noindent\textit{Notre environnement (air, eau, sol, aliments, corps humain) abrite une
multitude d'êtres vivants invisibles à l'œil nu : les microorganismes. La plupart sont
inoffensifs, voire utiles, mais certains peuvent provoquer des maladies. Comprendre
comment ils pénètrent dans l'organisme est la première étape pour s'en protéger.}
\vspace{8pt}

\connecteBanner

\section{Les microorganismes de notre environnement}

\begin{definition}[Microorganisme]
Un \textbf{microorganisme} (ou micro-être vivant) est un être vivant de très petite
taille, invisible à l'œil nu, observable seulement au microscope.
\end{definition}

\begin{propriete}[Quatre grands groupes]
\begin{itemize}
\item les \textbf{bactéries} : cellules très simples, sans noyau individualisé ;
\item les \textbf{virus} : particules encore plus petites, incapables de se
multiplier seules (elles ont besoin d'infecter une cellule hôte) ;
\item les \textbf{champignons microscopiques} (moisissures, levures) ;
\item certains \textbf{unicellulaires} (organismes formés d'une seule cellule, comme
certains parasites).
\end{itemize}
\end{propriete}

\begin{propriete}[Omniprésents dans l'environnement]
Les microorganismes sont présents partout : dans l'air, l'eau, le sol, les aliments, et
à la surface comme à l'intérieur du corps humain (peau, bouche, tube digestif\dots).
\end{propriete}

\begin{remarque}
La grande majorité des microorganismes sont \textbf{inoffensifs}, et beaucoup sont même
\textbf{utiles} : certaines bactéries de notre tube digestif aident à la digestion,
certaines levures servent à fabriquer le pain ou la bière. Seule une minorité de
microorganismes, dits \textbf{pathogènes}, peut provoquer des maladies.
\end{remarque}

\begin{exemple}
Les levures utilisées en boulangerie et certaines bactéries du yaourt sont des
microorganismes utiles ; le bacille de la tuberculose est un exemple de bactérie
pathogène.
\end{exemple}

\section{La contamination : comment les microorganismes pénètrent dans l'organisme}

\begin{definition}[Contamination]
La \textbf{contamination} est la pénétration d'un microorganisme pathogène dans
l'organisme. Elle précède l'\textbf{infection}, c'est-à-dire le développement de ce
microorganisme dans le corps et l'apparition éventuelle de symptômes.
\end{definition}

\begin{propriete}[Les principales voies de contamination]
\begin{itemize}
\item \textbf{voie cutanée} : par une plaie, une piqûre d'insecte, une morsure ;
\item \textbf{voie respiratoire} : par inhalation de microorganismes présents dans
l'air (toux, éternuement d'une personne malade) ;
\item \textbf{voie digestive} : par ingestion d'eau ou d'aliments contaminés ;
\item \textbf{voie sexuelle} : lors de rapports sexuels non protégés ;
\item \textbf{voie sanguine} : par contact avec du sang contaminé (matériel médical
non stérile, transfusion non contrôlée).
\end{itemize}
\end{propriete}

\begin{exemple}
Boire de l'eau non traitée dans une zone où l'assainissement est insuffisant expose à
une contamination par voie digestive (bactéries responsables de diarrhées, par
exemple).
\end{exemple}

\begin{remarque}
Piège fréquent : contamination et infection ne sont pas synonymes. Une personne peut
être contaminée (le microorganisme est entré dans son corps) sans être encore malade :
l'infection, avec ses symptômes, peut apparaître seulement après un certain temps
(période d'incubation).
\end{remarque}

\section{Une multiplication souvent très rapide}

\begin{propriete}[Reproduction rapide des bactéries]
De nombreuses bactéries se reproduisent par \textbf{scissiparité} : une bactérie se
divise en deux cellules filles identiques, qui se divisent à leur tour. Dans des
conditions favorables (chaleur, humidité, nutriments), cette multiplication peut être
très rapide, ce qui explique la vitesse à laquelle une contamination peut évoluer en
infection si l'organisme ne parvient pas à s'en défendre (voir chapitre suivant).
\end{propriete}

\begin{remarque}
C'est cette multiplication rapide qui justifie les gestes d'hygiène simples (lavage des
mains, conservation des aliments au froid, cuisson suffisante) : limiter le nombre de
microorganismes présents au départ retarde ou empêche le développement d'une
infection.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Microorganisme.} Être vivant microscopique : bactérie, virus, champignon
microscopique, ou unicellulaire.\\[4pt]
\textbf{Majorité inoffensive.} Beaucoup de microorganismes sont utiles ; une minorité
seulement est pathogène.\\[4pt]
\textbf{Contamination.} Pénétration d'un microorganisme pathogène dans
l'organisme.\\[4pt]
\textbf{Cinq voies.} Cutanée, respiratoire, digestive, sexuelle, sanguine.\\[4pt]
\textbf{Infection.} Développement du microorganisme dans le corps, avec apparition
possible de symptômes, après la contamination.\\[4pt]
\textbf{Piège fréquent.} Contamination $\neq$ infection : on peut être contaminé sans
présenter (encore) de symptômes.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Identifier une voie de contamination}
Repérer, dans une situation donnée, comment le microorganisme est entré en contact
avec l'organisme (plaie, air inhalé, aliment ingéré, rapport sexuel, sang) pour nommer
la voie correspondante.
\begin{exemple}
Une toux qui projette des gouttelettes contaminées correspond à une contamination par
voie respiratoire.
\end{exemple}

\methode{Distinguer contamination et infection}
Vérifier si des symptômes sont déjà présents (infection) ou si seul le contact avec le
microorganisme est mentionné (contamination).
\begin{exemple}
Une personne qui a marché pieds nus sur une plaie ouverte est contaminée ; si la plaie
s'infecte (rougeur, douleur, fièvre), on parle alors d'infection.
\end{exemple}
