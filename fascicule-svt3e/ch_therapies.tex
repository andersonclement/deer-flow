\chapter{Aider le système immunitaire}

\noindent\textit{Face à un microbe, la réponse spécifique du chapitre 7 met plusieurs
jours à devenir efficace. La médecine dispose de plusieurs moyens pour renforcer ou
accélérer cette défense : les antibiotiques, la sérothérapie et la vaccination.}
\vspace{8pt}

\connecteBanner

\section{L'antibiothérapie}

\begin{definition}[Antibiotique]
Un \textbf{antibiotique} est une substance qui \textbf{tue} ou \textbf{empêche la
multiplication} des bactéries. L'\textbf{antibiothérapie} est le traitement d'une
infection bactérienne par un antibiotique.
\end{definition}

\begin{propriete}[Une action ciblée sur les bactéries]
Les antibiotiques agissent uniquement sur les \textbf{bactéries} : ils sont
\textbf{inefficaces} contre les infections dues à des virus (comme un simple rhume).
\end{propriete}

\begin{remarque}
L'utilisation excessive ou incorrecte des antibiotiques favorise l'apparition de
bactéries \textbf{résistantes}, contre lesquelles ces médicaments ne fonctionnent
plus. C'est pourquoi un traitement antibiotique doit toujours être pris en respectant
la dose et la durée prescrites par un professionnel de santé.
\end{remarque}

\section{La sérothérapie}

\begin{definition}[Sérothérapie]
La \textbf{sérothérapie} consiste à injecter directement à un patient un
\textbf{sérum} contenant des \textbf{anticorps déjà fabriqués} (par un animal ou un
autre être humain), dirigés contre un microbe ou une toxine précise.
\end{definition}

\begin{propriete}[Une protection immédiate mais courte]
Comme les anticorps sont injectés tout prêts, la sérothérapie agit
\textbf{immédiatement}. Mais ces anticorps finissent par disparaître de l'organisme :
la protection est donc \textbf{de courte durée} (quelques semaines).
\end{propriete}

\begin{exemple}
Une personne mordue par un serpent venimeux reçoit un sérum antivenimeux : les
anticorps injectés neutralisent immédiatement le venin, sans attendre que l'organisme
fabrique lui-même ses propres anticorps.
\end{exemple}

\section{La vaccinothérapie}

\begin{definition}[Vaccination]
La \textbf{vaccination} consiste à injecter une forme \textbf{inactivée} ou
\textbf{atténuée} d'un microbe (ou de l'une de ses toxines), incapable de provoquer la
maladie, mais capable de déclencher la réponse spécifique de l'organisme (voir
chapitre 7).
\end{definition}

\begin{propriete}[Une protection différée mais durable]
Après la vaccination, l'organisme fabrique lui-même ses anticorps et surtout des
\textbf{cellules mémoires}. La protection n'est donc pas immédiate (elle nécessite
plusieurs jours à plusieurs semaines pour se mettre en place), mais elle est
\textbf{durable}, parfois pendant plusieurs années.
\end{propriete}

\begin{exemple}
Le vaccin contre le tétanos, administré avant tout contact avec la bactérie
responsable, prépare l'organisme à réagir rapidement et efficacement en cas
d'infection future.
\end{exemple}

\begin{propriete}[Sérothérapie et vaccination : deux logiques complémentaires]
\begin{center}
\small
\begin{tabular}{lcc}
\toprule
& \textbf{Sérothérapie} & \textbf{Vaccination} \\
\midrule
Ce qui est injecté & anticorps tout prêts & microbe inactivé/atténué \\
Origine des anticorps & extérieure & fabriqués par l'organisme \\
Délai d'action & immédiat & différé (jours à semaines) \\
Durée de protection & courte & longue \\
Usage principal & curatif (urgence) & préventif \\
\bottomrule
\end{tabular}
\end{center}
\end{propriete}

\section{La séro-vaccinothérapie}

\begin{definition}[Séro-vaccinothérapie]
La \textbf{séro-vaccinothérapie} associe, lors d'une même prise en charge, une
\textbf{sérothérapie} (protection immédiate) et une \textbf{vaccination} (protection
durable), pour couvrir à la fois le court terme et le long terme.
\end{definition}

\begin{exemple}
Une personne non vaccinée, blessée et exposée à un risque de tétanos, reçoit à la fois
un sérum antitétanique (protection immédiate le temps que l'infection ne se
déclare pas) et une injection de vaccin (protection durable pour l'avenir).
\end{exemple}

\begin{remarque}
Piège fréquent : ne pas confondre sérothérapie et vaccination. Dans la sérothérapie,
l'organisme \textbf{reçoit} des anticorps déjà fabriqués ; dans la vaccination,
l'organisme \textbf{fabrique lui-même} ses anticorps, après avoir été stimulé par le
vaccin.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Antibiothérapie.} Traitement d'une infection bactérienne par un antibiotique ;
inefficace contre les virus.\\[4pt]
\textbf{Sérothérapie.} Injection d'anticorps tout prêts : protection immédiate mais
courte, usage curatif.\\[4pt]
\textbf{Vaccination.} Injection d'un microbe inactivé/atténué : l'organisme fabrique
lui-même ses anticorps et ses cellules mémoires ; protection différée mais durable,
usage préventif.\\[4pt]
\textbf{Séro-vaccinothérapie.} Association des deux : protection immédiate et
durable.\\[4pt]
\textbf{Piège fréquent.} Sérothérapie = anticorps reçus tout prêts ; vaccination =
anticorps fabriqués par l'organisme lui-même.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Distinguer sérothérapie et vaccination}
Se demander si les anticorps sont directement injectés (sérothérapie) ou si
l'organisme est stimulé pour les fabriquer lui-même (vaccination), et comparer le
délai et la durée de protection.
\begin{exemple}
Un sérum antivenimeux agit immédiatement mais brièvement (sérothérapie) ; un vaccin
protège durablement mais après un délai (vaccination).
\end{exemple}

\item Définis l'antibiothérapie et précise contre quels agents pathogènes elle est

\item Définis la sérothérapie.
\item Définis la vaccination.
\item Compare, en un tableau ou en deux phrases, le délai d'action et la durée de

dans la vaccination.
\end{enumerate}

\section{Corrigés}
\begin{multicols}{2}

\corrige{de l'exercice 1}
Une substance qui tue ou empêche la multiplication des bactéries.

\corrige{de l'exercice 2}
Non, car les antibiotiques agissent uniquement sur les bactéries, pas sur les virus.

\corrige{de l'exercice 3}
Un sérum contenant des anticorps déjà fabriqués.

\corrige{de l'exercice 4}
Car ces anticorps injectés, d'origine extérieure, finissent par disparaître de
l'organisme.

\corrige{de l'exercice 5}
Une forme inactivée ou atténuée d'un microbe ou de l'une de ses toxines.

\corrige{de l'exercice 6}
Car l'organisme doit d'abord fabriquer lui-même ses anticorps et ses cellules
mémoires, ce qui prend du temps.

\corrige{de l'exercice 7}
Une sérothérapie (protection immédiate) et une vaccination (protection durable).

\corrige{de l'exercice 8}
Car un arrêt précoce peut laisser survivre des bactéries partiellement résistantes,
favorisant l'apparition de bactéries résistantes à l'antibiotique.

\corrige{de l'exercice 9}
Dans la sérothérapie, les anticorps sont d'origine extérieure, injectés tout prêts ;
dans la vaccination, les anticorps sont fabriqués par l'organisme lui-même, après
stimulation par le vaccin.

\corrige{de l'exercice 10}
Car la sérothérapie agit immédiatement, ce qui est indispensable en cas d'urgence, alors
que la vaccination nécessite un délai avant d'être efficace.

\corrige{de l'exercice 11}
Car les anticorps injectés lors de la sérothérapie finissent par disparaître, alors
que la vaccination laisse des cellules mémoires durables dans l'organisme.

\corrige{de l'exercice 12}
Parce que la personne n'étant pas vaccinée, elle n'a pas de protection immédiate ; la
sérothérapie protège dans l'urgence le temps que la vaccination mette en place une
protection durable pour l'avenir.

\corrige{du sujet 1}
1) L'antibiothérapie est le traitement d'une infection bactérienne par un
antibiotique ; elle n'est efficace que contre les bactéries, pas contre les virus.\\
2) La sérothérapie est l'injection d'anticorps déjà fabriqués contre un microbe ou une
toxine précise.\\
3) La vaccination est l'injection d'un microbe inactivé ou atténué, qui stimule
l'organisme à fabriquer lui-même ses anticorps et ses cellules mémoires.\\
4) La sérothérapie agit immédiatement mais brièvement ; la vaccination agit après un
délai mais durablement.

\corrige{du sujet 2}
1) L'association, lors d'une même prise en charge, d'une sérothérapie et d'une
vaccination.\\
2) Pour bénéficier à la fois d'une protection immédiate (sérothérapie) et d'une
protection durable (vaccination).\\
3) Par exemple : une personne non vaccinée exposée à un risque de tétanos après une
blessure.\\
4) Dans la sérothérapie, les anticorps viennent de l'extérieur (déjà fabriqués) ;
dans la vaccination, ils sont fabriqués par l'organisme lui-même.

\end{multicols}
