\chapter{L'hérédité et les anomalies chromosomiques}

\noindent\textit{Pourquoi ressemble-t-on à ses parents, sans leur être identique ? La
réponse se trouve dans nos chromosomes, qui portent l'information génétique transmise de
génération en génération. Ce chapitre étudie cette transmission, et le cas où elle se
déroule anormalement.}
\vspace{8pt}

\connecteBanner

\section{Chromosomes, gènes et allèles}

\begin{definition}[Chromosome]
Un \textbf{chromosome} est une structure du noyau cellulaire qui porte l'information
génétique. L'espèce humaine possède \textbf{46 chromosomes}, organisés en \textbf{23
paires} : \textbf{22 paires d'autosomes} (identiques chez l'homme et la femme) et
\textbf{1 paire de chromosomes sexuels} (XX chez la femme, XY chez l'homme).
\end{definition}

\begin{propriete}[Le caryotype]
Le \textbf{caryotype} est la représentation ordonnée de l'ensemble des chromosomes d'un
individu, classés par paires de taille décroissante. Il permet de vérifier le nombre et
la structure des chromosomes.
\end{propriete}

\begin{definition}[Gène et allèle]
Un \textbf{gène} est une portion de chromosome qui porte une information héréditaire
précise (par exemple, la couleur des yeux). Un \textbf{allèle} est une version
particulière de ce gène. Chaque individu possède \textbf{deux allèles} pour un gène donné
(un hérité de chaque parent), situés au même endroit sur les deux chromosomes d'une
paire.
\end{definition}

\begin{propriete}[Allèle dominant, allèle récessif]
Un allèle \textbf{dominant} s'exprime dès qu'il est présent, même en un seul exemplaire.
Un allèle \textbf{récessif} ne s'exprime que si les \textbf{deux} allèles portés par
l'individu sont récessifs.
\end{propriete}

\section{Transmission d'un caractère héréditaire}

\begin{propriete}[Un exemple : la couleur des yeux]
On note $B$ l'allèle « yeux bruns » (dominant) et $b$ l'allèle « yeux bleus »
(récessif). Un individu $Bb$ (un allèle de chaque) a les \textbf{yeux bruns} : seul un
individu $bb$ (deux allèles récessifs) a les \textbf{yeux bleus}.
\end{propriete}

\begin{illus}
\begin{center}
\small
\begin{tabular}{c|cc}
& $B$ & $b$ \\
\hline
$B$ & $BB$ & $Bb$ \\
$b$ & $Bb$ & $bb$ \\
\end{tabular}
\end{center}
\fig{Figure — Échiquier de croisement : deux parents $Bb$ (yeux bruns) donnent des enfants $BB$, $Bb$ ou $bb$.}
\end{illus}

\begin{propriete}[Résultat du croisement $Bb\times Bb$]
Sur $4$ combinaisons possibles : $1$ enfant $BB$, $2$ enfants $Bb$, $1$ enfant $bb$, soit
les proportions $\dfrac14$, $\dfrac12$, $\dfrac14$. Comme $B$ est dominant, $3$ enfants
sur $4$ (les $BB$ et les $Bb$) ont les \textbf{yeux bruns}, et $1$ sur $4$ (le $bb$) a les
\textbf{yeux bleus}.
\end{propriete}

\begin{exemple}
Deux parents aux yeux bruns, tous deux $Bb$, peuvent avoir un enfant aux yeux bleus
($bb$) : chacun leur a transmis, par hasard, son allèle récessif $b$.
\end{exemple}

\section{Les anomalies chromosomiques}

\begin{definition}[Anomalie chromosomique]
Une \textbf{anomalie chromosomique} est une modification du nombre ou de la structure
normale des chromosomes, visible sur un caryotype.
\end{definition}

\begin{propriete}[La trisomie 21]
La \textbf{trisomie 21} (syndrome de Down) est due à la présence de \textbf{trois}
chromosomes 21 au lieu de deux : le caryotype comporte donc $47$ chromosomes au lieu de
$46$. Cette anomalie apparaît accidentellement lors de la formation des cellules
reproductrices et entraîne des particularités physiques et un développement
intellectuel spécifique chez la personne concernée.
\end{propriete}

\begin{remarque}
Le \textbf{dépistage prénatal} (échographie, prise de sang, caryotype fœtal) permet de
détecter certaines anomalies chromosomiques avant la naissance, offrant un accompagnement
médical adapté aux futurs parents.
\end{remarque}

\section{L'essentiel à retenir}

\begin{synthese}
\textbf{Chromosomes.} $46$ chromosomes humains $=23$ paires ($22$ autosomes $+1$ paire
sexuelle).\\[4pt]
\textbf{Gène / allèle.} Un gène = une information ; un allèle = une version de cette
information ; deux allèles par individu.\\[4pt]
\textbf{Dominant / récessif.} Le dominant s'exprime dès qu'il est présent ; le récessif
seulement à l'état homozygote.\\[4pt]
\textbf{Croisement $Bb\times Bb$.} Proportions $\dfrac14\,BB$, $\dfrac12\,Bb$,
$\dfrac14\,bb$, soit $\dfrac34$ phénotype dominant, $\dfrac14$ phénotype récessif.\\[4pt]
\textbf{Trisomie 21.} $3$ chromosomes 21 au lieu de $2$, soit $47$ chromosomes au
total.\\[4pt]
\textbf{Piège fréquent.} Deux parents au phénotype dominant peuvent transmettre un
phénotype récessif à leur enfant s'ils sont tous deux hétérozygotes.
\end{synthese}

\section{Méthodes \& savoir-faire}

\methode{Construire un échiquier de croisement}
Placer les allèles d'un parent en ligne, ceux de l'autre en colonne, puis remplir chaque
case avec la combinaison correspondante.
\begin{exemple}
Pour $Bb\times Bb$, les quatre cases donnent $BB$, $Bb$, $Bb$, $bb$.
\end{exemple}

\methode{Déterminer un phénotype à partir d'un génotype}
Un individu portant au moins un allèle dominant a le phénotype dominant ; seul
l'homozygote récessif a le phénotype récessif.
\begin{exemple}
$Bb$ et $BB$ ont tous deux les yeux bruns ; seul $bb$ a les yeux bleus.
\end{exemple}

\methode{Reconnaître une anomalie chromosomique sur un caryotype}
Compter le nombre de chromosomes d'une paire : deux chromosomes = normal ; trois
chromosomes = trisomie.
\begin{exemple}
Un caryotype à $47$ chromosomes, avec trois chromosomes 21, indique une trisomie 21.
\end{exemple}
